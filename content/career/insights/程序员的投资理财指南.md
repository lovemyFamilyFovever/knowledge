## 一、财务自由的基本概念：代码世界的财务解耦

对于程序员而言，财务自由可以类比为**系统架构中的解耦**。当你的投资组合产生的被动收入能够覆盖生活开支时，你就实现了与工作收入的“解耦”。

**财务自由的三个关键指标：**
1. **财务安全线**：被动收入 ≥ 基本生活开支（约当地平均工资的1.5倍）
2. **财务舒适线**：被动收入 ≥ 当前税后工资收入
3. **完全财务自由**：被动收入 ≥ 理想生活状态开支（通常为当前开支的2-3倍）

**案例分析：**
假设小王月开支1.5万元，年薪50万（税后约35万）。他需要：
- 财务安全线：1.5万×12×25 = 450万（按4%安全提取率计算）
- 财务舒适线：35万÷4% = 875万投资组合
- 完全自由：1.5万×2×12÷3.5% ≈ 1028万（考虑3.5%提取率更保守）

```python
# 财务自由计算函数
def calculate_fire_number(annual_expense, withdrawal_rate=0.04):
    """
    计算财务自由所需本金
    :param annual_expense: 年度开支（税后）
    :param withdrawal_rate: 安全提取率，通常4%
    :return: 所需投资组合总额
    """
    fire_number = annual_expense / withdrawal_rate
    return fire_number

# 使用示例
```

```python
# 预算管理自动化脚本
class BudgetManager:
    def __init__(self, monthly_income):
        self.income = monthly_income
        self.categories = {
            'necessities': 0.5,
            'discretionary': 0.3,
            'savings_investments': 0.2
        }
        self.actual = {cat: 0 for cat in self.categories}
    
    def track_expense(self, category, amount):
        """记录实际开支"""
        if category in self.actual:
            self.actual[category] += amount
            self.check_budget(category)
    
    def check_budget(self, category):
        """检查预算执行情况"""
        budget = self.income * self.categories[category]
        actual = self.actual[category]
        percentage = (actual / budget) * 100
        
        if percentage > 100:
            print(f"⚠️ {category}超支：{actual:.0f}/{budget:.0f}")
        elif percentage > 80:
            print(f"📈 {category}接近预算：{actual:.0f}/{budget:.0f}")
    
    def generate_report(self):
        """生成月度报告"""
        report = {}
        for category in self.categories:
            budget = self.income * self.categories[category]
            actual = self.actual[category]
            report[category] = {
                'budget': budget,
                'actual': actual,
                'variance': actual - budget
            }
        return report

# 使用示例
income = 30000  # 月收入3万
budget_mgr = BudgetManager(income)
budget_mgr.track_expense('necessities', 8000)  # 房租水电
```

```python
import pandas as pd
from datetime import datetime

class EmergencyFundBuilder:
    def __init__(self, target_amount, monthly_contribution):
        self.target = target_amount
        self.monthly = monthly_contribution
        self.current = 0
        self.contributions = []
    
    def make_contribution(self, date=None):
        if date is None:
            date = datetime.now().strftime("%Y-%m")
        
        self.current += self.monthly
        self.contributions.append({
            'date': date,
            'amount': self.monthly,
            'total': self.current
        })
        
        progress = (self.current / self.target) * 100
        print(f"✅ {date}: 存入{self.monthly}元，累计{self.current}元 ({progress:.1f}%)")
    
    def check_milestone(self):
        milestones = [0.25, 0.5, 0.75, 1.0]
        for m in milestones:
            if self.current >= self.target * m:
                print(f"🎯 达成里程碑：{m*100}%！")
    
    def simulate_months(self, months):
        """模拟存入过程"""
        print(f"模拟{months}个月的应急基金构建：")
        for i in range(months):
            self.make_contribution()
            self.check_milestone()
```

### 4.2 复利效应：程序员的指数增长

**复利公式**：A = P × (1 + r/n)^(n×t)
其中：
- A = 最终金额
- P = 本金
- r = 年利率
- n = 每年复利次数
- t = 年数

```python
# 复利计算器
def compound_interest(principal, rate, years, compounding='yearly'):
    """
    计算复利收益
    :param principal: 本金
    :param rate: 年利率（如0.07表示7%）
    :param years: 年数
    :param compounding: 复利频率 daily/monthly/yearly
    :return: 最终金额和收益
    """
    freq_map = {
        'daily': 365,
        'monthly': 12,
        'quarterly': 4,
        'yearly': 1
    }
    
    n = freq_map[compounding]
    total = principal * (1 + rate/n) ** (n * years)
    profit = total - principal
    
    return {
        'final_amount': total,
        'profit': profit,
        'effective_rate': (total/principal) ** (1/years) - 1
    }

# 不同场景下的复利效应
scenarios = [
    {'name': '保守投资', 'principal': 100000, 'rate': 0.04, 'years': 20},
```

### 4.3 资产配置：投资组合的架构设计

**核心原则**：
1. **分散化**：不把所有鸡蛋放在一个篮子里
2. **相关性**：选择相关性低的资产
3. **再平衡**：定期调整回目标比例

```python
# 简单的资产配置模拟器
class PortfolioAllocator:
    def __init__(self, initial_investment):
        self.initial = initial_investment
        self.portfolio = {}
        self.target_allocation = {}
        self.actual_allocation = {}
    
    def set_target_allocation(self, allocation):
        """
        设置目标资产配置
        :param allocation: 资产类别和比例的字典，如 {'stocks': 0.6, 'bonds': 0.4}
        """
        self.target_allocation = allocation
        # 初始化实际持仓
        for asset, ratio in allocation.items():
            self.portfolio[asset] = {
                'value': self.initial * ratio,
                'shares': 0,  # 需要后续计算
                'price': 1
            }
        self.update_actual_allocation()
    
    def update_actual_allocation(self):
        """更新实际资产配置比例"""
        total = sum([v['value'] for v in self.portfolio.values()])
        for asset in self.portfolio:
            self.actual_allocation[asset] = self.portfolio[asset]['value'] / total
    
    def rebalance(self):
        """执行再平衡"""
        total_value = sum([v['value'] for v in self.portfolio.values()])
        
        print("开始再平衡操作：")
        for asset, target_ratio in self.target_allocation.items():
            current_value = self.portfolio[asset]['value']
            target_value = total_value * target_ratio
            difference = target_value - current_value
            
            if abs(difference) > 100:  # 只调整差异大于100元的部分
                action = "买入" if difference > 0 else "卖出"
                print(f"  {action} {asset}：调整{abs(difference):.2f}元")
                self.portfolio[asset]['value'] = target_value
        
        self.update_actual_allocation()
        print("再平衡完成！")
    
    def simulate_year(self, returns):
        """
        模拟一年的投资收益
        :param returns: 各资产类别的收益率字典
        """
        for asset, ret in returns.items():
            if asset in self.portfolio:
                self.portfolio[asset]['value'] *= (1 + ret)
        self.update_actual_allocation()

# 使用示例
portfolio = PortfolioAllocator(100000)
portfolio.set_target_allocation({
    'A股指数': 0.4,
    '美股指数': 0.2,
    '债券': 0.3,
    '黄金': 0.1
})

# 模拟各资产一年收益率
returns = {
    'A股指数': 0.08,
    '美股指数': 0.12,
    '债券': 0.04,
    '黄金': 0.02
}

portfolio.simulate_year(returns)
print(f"年末实际配置：{portfolio.actual_allocation}")
```

```python
# 指数基金筛选器
class IndexFundScreener:
    def __init__(self):
        self.criteria = {
            'tracking_error': 0.02,  # 跟踪误差阈值
            'expense_ratio': 0.005,  # 管理费率阈值
            'fund_size': 10,  # 基金规模阈值（亿元）
            'history_years': 3  # 成立年限阈值
        }
    
    def screen_funds(self, funds_data):
        """筛选符合条件的指数基金"""
        qualified = []
        
        for fund in funds_data:
            if (fund.get('tracking_error', 999) <= self.criteria['tracking_error'] and
                fund.get('expense_ratio', 999) <= self.criteria['expense_ratio'] and
                fund.get('fund_size', 0) >= self.criteria['fund_size'] and
                fund.get('history_years', 0) >= self.criteria['history_years']):
                qualified.append(fund)
        
        return sorted(qualified, key=lambda x: x.get('expense_ratio', 999))
    
    def compare_funds(self, fund_list):
        """比较基金的各项指标"""
        comparison = {}
        for fund in fund_list:
            name = fund.get('name', 'Unknown')
            comparison[name] = {
                '费率': f"{fund.get('expense_ratio', 0)*100:.2f}%",
                '规模': f"{fund.get('fund_size', 0):.1f}亿",
                '跟踪误差': f"{fund.get('tracking_error', 0)*100:.3f}%",
                '成立年限': f"{fund.get('history_years', 0)}年"
            }
        
        return comparison
```

### 5.2 价值投资：寻找被低估的优质公司

**价值投资核心指标：**
1. **市盈率（P/E）**：股价/每股收益
2. **市净率（P/B）**：股价/每股净资产
3. **股息率**：每股股息/股价
4. **自由现金流**：经营活动现金流-资本开支

```python
# 价值投资评分系统
class ValueInvestor:
    def __init__(self):
        self.weights = {
            'pe_ratio': 0.25,
            'pb_ratio': 0.20,
            'dividend_yield': 0.15,
            'debt_to_equity': 0.15,
            'current_ratio': 0.10,
            'roe': 0.15
        }
    
    def calculate_score(self, company_data):
        """
        计算公司价值投资评分
        :param company_data: 公司财务数据字典
        :return: 综合评分
        """
        score = 0
        
        # 市盈率评分（越低越好，但要考虑行业平均）
        pe = company_data.get('pe_ratio', 0)
        if pe > 0 and pe < 15:
            score += self.weights['pe_ratio'] * 100
        elif pe >= 15 and pe < 25:
            score += self.weights['pe_ratio'] * 70
        
        # 市净率评分
        pb = company_data.get('pb_ratio', 0)
        if pb > 0 and pb < 1.5:
            score += self.weights['pb_ratio'] * 100
        elif pb >= 1.5 and pb < 3:
            score += self.weights['pb_ratio'] * 60
        
        # 股息率评分
        dividend = company_data.get('dividend_yield', 0)
        if dividend > 0.03:
            score += self.weights['dividend_yield'] * 100
        elif dividend > 0.02:
            score += self.weights['dividend_yield'] * 70
        
        # ROE评分（净资产收益率）
        roe = company_data.get('roe', 0)
        if roe > 0.15:
            score += self.weights['roe'] * 100
        elif roe > 0.10:
            score += self.weights['roe'] * 60
        
        return score
    
    def find_undervalued(self, companies):
        """寻找被低估的公司"""
        undervalued = []
        
        for company in companies:
            score = self.calculate_score(company)
            if score >= 70:  # 设定阈值
                company['score'] = score
                undervalued.append(company)
        
        return sorted(undervalued, key=lambda x: x['score'], reverse=True)

    },

        'name': '公司B',
```

```python
# 简单的移动平均线策略
import pandas as pd
import numpy as np

class MovingAverageStrategy:
    def __init__(self, short_window=20, long_window=50):
        self.short_window = short_window
        self.long_window = long_window
    
    def generate_signals(self, price_data):
        """
        生成交易信号
        :param price_data: 价格序列
        :return: 包含信号的DataFrame
        """
        signals = pd.DataFrame(index=price_data.index)
        signals['price'] = price_data
        
        # 计算移动平均线
        signals['short_ma'] = price_data.rolling(window=self.short_window).mean()
        signals['long_ma'] = price_data.rolling(window=self.long_window).mean()
        
        # 生成信号：短期均线上穿长期均线买入，下穿卖出
        signals['signal'] = 0
        signals.loc[signals['short_ma'] > signals['long_ma'], 'signal'] = 1
        signals.loc[signals['short_ma'] < signals['long_ma'], 'signal'] = -1
        
        # 计算持仓变化
        signals['positions'] = signals['signal'].diff()
        
        return signals
    
    def backtest(self, price_data, initial_capital=100000):
        """
        回测策略
        :param price_data: 价格数据
        :param initial_capital: 初始资金
        :return: 回测结果
        """
        signals = self.generate_signals(price_data)
        
        # 模拟交易
        capital = initial_capital
        positions = 0
        portfolio_value = []
        
        for i in range(len(signals)):
            current_price = signals['price'].iloc[i]
            
            # 买入信号
            if signals['positions'].iloc[i] == 1 and capital > 0:
                shares_to_buy = capital // current_price
                positions += shares_to_buy
                capital -= shares_to_buy * current_price
            
            # 卖出信号
            elif signals['positions'].iloc[i] == -1 and positions > 0:
                capital += positions * current_price
                positions = 0
            
            # 计算总价值
            total_value = capital + positions * current_price
            portfolio_value.append(total_value)
        
        signals['portfolio_value'] = portfolio_value
        
        # 计算收益指标
        returns = signals['portfolio_value'].pct_change().dropna()
        
        return {
            'final_value': signals['portfolio_value'].iloc[-1],
            'total_return': (signals['portfolio_value'].iloc[-1] / initial_capital - 1) * 100,
            'annualized_return': ((signals['portfolio_value'].iloc[-1] / initial_capital) ** (252/len(signals)) - 1) * 100,
            'max_drawdown': self.calculate_max_drawdown(signals['portfolio_value']),
            'sharpe_ratio': self.calculate_sharpe_ratio(returns)
        }
    
    def calculate_max_drawdown(self, portfolio_values):
        """计算最大回撤"""
        peak = portfolio_values.expanding(min_periods=1).max()
        drawdown = (portfolio_values - peak) / peak
        return drawdown.min() * 100
    
    def calculate_sharpe_ratio(self, returns, risk_free_rate=0.02/252):
        """计算夏普比率"""
        if returns.std() == 0:
            return 0
        excess_returns = returns - risk_free_rate
        return np.sqrt(252) * excess_returns.mean() / returns.std()
```

```python
                        metric_score = 70
                    else:

                        metric_score = 100
                    elif value < 0.005:
                        metric_score = 60
                    else:
                        metric_score = 30

                    # 默认评分逻辑
                    metric_score = 50

            }
```
