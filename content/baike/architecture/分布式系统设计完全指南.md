## 分布式系统设计完全指南

## 1) CAP定理和PACELC理论

### 1.1 CAP定理详解

CAP定理是分布式系统设计的基石，由Eric Brewer在2000年提出。它指出一个分布式系统最多只能同时满足以下三个特性中的两个：

- **一致性（Consistency）**：所有节点在同一时间看到相同的数据
- **可用性（Availability）**：每个请求都能在合理时间内得到响应
- **分区容错性（Partition Tolerance）**：系统在网络分区（节点间通信中断）的情况下仍能继续运行

```java
// CAP定理示例：银行转账系统
public class BankTransferService {
    // CP系统示例：确保强一致性，可能牺牲可用性
    public boolean transferCP(String fromAccount, String toAccount, double amount) {
        // 1. 获取分布式锁，确保串行化
        lock(fromAccount, toAccount);
        
        try {
            // 2. 检查余额（可能因网络分区返回失败）
            double balance = queryBalance(fromAccount);
            if (balance < amount) {
                return false;
            }
            
            // 3. 执行转账（确保所有节点同步更新）
            updateBalance(fromAccount, -amount);
            updateBalance(toAccount, amount);
            
            // 4. 等待所有副本确认
            waitForAllReplicasConfirm();
            
            return true;
        } finally {
            unlock(fromAccount, toAccount);
        }
    }
    
    // AP系统示例：确保可用性，可能牺牲强一致性
    public boolean transferAP(String fromAccount, String toAccount, double amount) {
        // 1. 尝试更新主节点（如果主节点不可用，选择其他节点）
        Node master = selectMasterNode(fromAccount);
        if (master == null || !master.isAvailable()) {
            master = selectAlternativeNode(fromAccount);
        }
        
        // 2. 执行本地事务（立即返回成功）
        asyncUpdateBalance(master, fromAccount, -amount);
        asyncUpdateBalance(master, toAccount, amount);
        
        // 3. 异步复制到其他节点
        asyncReplicateToOtherNodes(fromAccount, toAccount, amount);
        
        return true; // 立即返回，不保证强一致性
    }
}
```

### 1.2 PACELC理论

PACELC理论是CAP的扩展，由Daniel Abadi在2012年提出。它指出：在**分区（P）** 发生时，系统必须在**可用性（A）** 和**一致性（C）** 之间做出选择；在**正常（E）** 运行时，系统需要在**延迟（L）** 和**一致性（C）** 之间做出权衡。

```python
# PACELC决策模型示例
class PACELCDecisionEngine:
    def __init__(self, system_type):
        self.system_type = system_type  # 'pa_ec' 或 'pc_ec'
        
    def make_decision(self, is_partition, latency_requirement, consistency_requirement):
        if is_partition:
            # 分区发生时的决策
            if self.system_type == 'pa_ec':
                return self.choose_availability()
            else:
                return self.choose_consistency()
        else:
            # 正常运行时的决策
            if latency_requirement == 'low':
                return self.choose_low_latency()
            else:
                return self.choose_high_consistency()
    
    def choose_availability(self):
        return {
            'strategy': 'eventual_consistency',
            'timeout': 'short',
            'retry': 'aggressive',
            'conflict_resolution': 'last_write_wins'
        }
    
    def choose_consistency(self):
        return {
            'strategy': 'linearizability',
            'timeout': 'long',
            'retry': 'careful',
            'conflict_resolution': 'consensus_based'
        }
```

### 1.3 实际案例分析

**案例1：ZooKeeper（PC/EC系统）**
- 选择一致性而非可用性
- 使用ZAB协议保证强一致性
- 在网络分区时可能拒绝写请求

**案例2：Cassandra（PA/EL系统）**
- 选择可用性和低延迟
- 使用最终一致性模型
- 提供可调一致性级别

**案例3：CockroachDB（PC/EC系统）**
- 选择一致性和低延迟（通过优化网络）
- 使用Multi-Raft协议
- 提供分布式ACID事务

## 2) 一致性模型

### 2.1 强一致性（线性一致性）

强一致性（线性一致性）是最严格的一致性模型，要求所有操作看起来像在单个节点上按顺序执行。

```java
// 使用ZooKeeper实现强一致性读写
public class StrongConsistencyService {
    private final ZooKeeper zk;
    private final String lockPath = "/locks";
    private final String dataPath = "/data";
    
    public String readStrongConsistent(String key) throws Exception {
        // 1. 获取分布式锁（确保写操作串行化）
        String lockNode = zk.create(lockPath + "/read-", 
            new byte[0], ZooDefs.Ids.OPEN_ACL_UNSAFE, 
            CreateMode.EPHEMERAL_SEQUENTIAL);
        
        // 2. 等待获取锁
        waitForLock(lockNode);
        
        try {
            // 3. 从主节点读取（确保看到最新写入）
            Stat stat = new Stat();
            byte[] data = zk.getData(dataPath + "/" + key, false, stat);
            
            // 4. 验证版本号（可选）
            verifyVersionConsistency(stat.getVersion());
            
            return new String(data);
        } finally {
            // 5. 释放锁
            zk.delete(lockNode, -1);
        }
    }
    
    public void writeStrongConsistent(String key, String value) throws Exception {
        // 1. 获取排他写锁
        String writeLock = acquireWriteLock(key);
        
        try {
            // 2. 执行原子写入
            Stat stat = zk.setData(dataPath + "/" + key, 
                value.getBytes(), -1);  // 使用版本号确保CAS
            
            // 3. 同步到所有副本
            syncToAllReplicas(key, value, stat.getVersion());
            
        } finally {
            // 4. 释放锁
            releaseWriteLock(writeLock);
        }
    }
}
```

### 2.2 最终一致性

最终一致性保证如果没有新的更新，最终所有副本都会收敛到相同的状态。

```python
# Dynamo风格最终一致性实现
class EventuallyConsistentStore:
    def __init__(self, replicas, read_quorum, write_quorum):
        self.replicas = replicas
        self.read_quorum = read_quorum
        self.write_quorum = write_quorum
        self.version_vectors = {}  # 向量时钟
        
    def put(self, key, value, client_id=None):
        # 1. 生成向量时钟版本
        version = self._create_version(client_id)
        
        # 2. 并行写入多个副本
        write_results = []
        for replica in random.sample(self.replicas, 
                                    min(self.write_quorum + 1, len(self.replicas))):
            try:
                result = replica.write(key, value, version)
                write_results.append(result)
            except Exception as e:
                continue
        
        # 3. 检查是否达到写法定人数
        if len(write_results) >= self.write_quorum:
            return {
                'success': True,
                'version': version,
                'consistent_nodes': len(write_results)
            }
        else:
            return {'success': False, 'error': 'Write quorum not met'}
    
    def get(self, key):
        # 1. 并行读取多个副本
        read_results = []
        for replica in random.sample(self.replicas, 
                                    min(self.read_quorum + 1, len(self.replicas))):
            try:
                result = replica.read(key)
                if result:
                    read_results.append(result)
            except:
                continue
        
        if not read_results:
            return None
        
        # 2. 检测并解决冲突
        return self._resolve_conflicts(read_results)
    
    def _resolve_conflicts(self, values):
        # 使用向量时钟解决冲突
        if len(values) == 1:
            return values[0]
        
        # 查找最新版本（基于向量时钟）
        latest = values[0]
        for value in values[1:]:
            if self._is_newer(value['version'], latest['version']):
                latest = value
        
        return latest
```

### 2.3 因果一致性

因果一致性保证所有因果相关的操作按顺序被观察到。

```go
// 因果一致性实现（使用向量时钟）
package causalconsistency

import (
	"sync"
	"time"
)

type CausalConsistencyStore struct {
	mu          sync.RWMutex
	store       map[string]ValueWithCausality
	vectorClock map[string]uint64 // 节点ID -> 逻辑时间
	nodeID      string
}

type ValueWithCausality struct {
	Value       string
	VectorClock map[string]uint64
	Timestamp   time.Time
}

func NewCausalStore(nodeID string) *CausalConsistencyStore {
	return &CausalConsistencyStore{
		store:       make(map[string]ValueWithCausality),
		vectorClock: map[string]uint64{nodeID: 0},
		nodeID:      nodeID,
	}
}

func (c *CausalConsistencyStore) Put(key, value string, causality map[string]uint64) {
	c.mu.Lock()
	defer c.mu.Unlock()

	// 1. 更新本地向量时钟
	for node, ts := range causality {
		if c.vectorClock[node] < ts {
			c.vectorClock[node] = ts
		}
	}
	c.vectorClock[c.nodeID]++

	// 2. 创建新版本
	newClock := make(map[string]uint64)
	for k, v := range c.vectorClock {
		newClock[k] = v
	}

	// 3. 存储值
	c.store[key] = ValueWithCausality{
		Value:       value,
		VectorClock: newClock,
		Timestamp:   time.Now(),
	}
}

func (c *CausalConsistencyStore) Get(key string) (string, map[string]uint64) {
	c.mu.RLock()
	defer c.mu.RUnlock()

	if v, ok := c.store[key]; ok {
		return v.Value, v.VectorClock
	}
	return "", nil
}

// 传播因果信息
func (c *CausalConsistencyStore) PropagateTo(other *CausalConsistencyStore, key string) {
	c.mu.RLock()
	defer c.mu.RUnlock()

	if val, ok := c.store[key]; ok {
		other.Put(key, val.Value, val.VectorClock)
	}
}
```

### 2.4 读写一致性（Read-Your-Writes）

读写一致性保证客户端总是能看到自己之前的写入。

```javascript
// 客户端会话实现读写一致性
class ReadYourWritesClient {
    constructor(serviceEndpoints, sessionId) {
        this.serviceEndpoints = serviceEndpoints;
        this.sessionId = sessionId || this.generateSessionId();
        this.writeVersions = new Map(); // key -> 最新写版本
    }
    
    async write(key, value) {
        // 1. 发送写请求
        const endpoint = this.selectEndpoint();
        const response = await fetch(`${endpoint}/write`, {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'X-Session-ID': this.sessionId
            },
            body: JSON.stringify({ key, value })
        });
        
        const result = await response.json();
        
        if (result.success) {
            // 2. 记录写版本
            this.writeVersions.set(key, {
                version: result.version,
                timestamp: Date.now(),
                endpoint: endpoint
            });
            
            return result;
        } else {
            throw new Error(result.error);
        }
    }
    
    async read(key) {
        // 1. 检查是否有该键的写版本
        const writeInfo = this.writeVersions.get(key);
        
        if (writeInfo) {
            // 2. 优先从写入节点读取（保证读到自己的写入）
            try {
                const response = await fetch(
                    `${writeInfo.endpoint}/read?key=${key}&version=${writeInfo.version}`
                );
                
                const result = await response.json();
                if (result.version >= writeInfo.version) {
                    return result.value;
                }
            } catch (error) {
                // 降级到其他节点
            }
        }
        
        // 3. 从任意可用节点读取
        const endpoint = this.selectEndpoint();
        const response = await fetch(`${endpoint}/read?key=${key}`);
        return response.json().value;
    }
    
    selectEndpoint() {
        // 负载均衡或一致性哈希选择
        return this.serviceEndpoints[
            Math.floor(Math.random() * this.serviceEndpoints.length)
        ];
    }
    
    generateSessionId() {
        return 'session_' + Date.now() + '_' + Math.random().toString(36).substr(2);
    }
}
```

## 3) 共识算法

### 3.1 Paxos算法

Paxos是分布式共识的基础算法，由Leslie Lamport提出。

```python
# Basic Paxos实现（简化版）
class PaxosNode:
    def __init__(self, node_id, peers):
        self.node_id = node_id
        self.peers = peers  # 其他节点
        self.proposal_number = 0
        self.accepted_proposal = None
        self.accepted_value = None
        self.promised_number = 0
        
    async def prepare(self, proposal_number):
        """Phase 1a: Prepare"""
        promises = []
        for peer in self.peers:
            try:
                promise = await peer.handle_prepare(proposal_number)
                promises.append(promise)
            except:
                continue
        
        # 检查是否收到多数派承诺
        if len(promises) > len(self.peers) / 2:
            # 选择最大的已接受提案
            accepted_proposals = [p for p in promises if p.get('accepted')]
            if accepted_proposals:
                max_proposal = max(accepted_proposals, 
                                 key=lambda x: x['proposal_number'])
                return max_proposal['value']
            return None  # 可以自由提议
        else:
            raise Exception("Failed to get majority promises")
    
    async def accept(self, proposal_number, value):
        """Phase 2a: Accept"""
        accepts = 0
        for peer in self.peers:
            try:
                result = await peer.handle_accept(proposal_number, value)
                if result['accepted']:
                    accepts += 1
            except:
                continue
        
        # 检查是否获得多数派接受
        if accepts > len(self.peers) / 2:
            return True
        else:
            return False
    
    async def handle_prepare(self, proposal_number):
        """Phase 1b: Promise"""
        if proposal_number > self.promised_number:
            self.promised_number = proposal_number
            return {
                'promised': True,
                'proposal_number': self.proposal_number,
                'accepted_value': self.accepted_value
            }
        else:
            return {'promised': False}
    
    async def handle_accept(self, proposal_number, value):
        """Phase 2b: Accepted"""
        if proposal_number >= self.promised_number:
            self.promised_number = proposal_number
            self.proposal_number = proposal_number
            self.accepted_value = value
            return {'accepted': True}
        else:
            return {'accepted': False}
    
    async def propose(self, value):
        """完整的Paxos提案过程"""
        self.proposal_number += 1
        
        # Phase 1: Prepare
        try:
            existing_value = await self.prepare(self.proposal_number)
        except Exception as e:
            return False
        
        # Phase 2: Accept
        if existing_value:
            value = existing_value  # 使用已接受的值
        
        success = await self.accept(self.proposal_number, value)
        
        if success:
            # Phase 3: Learn (广播结果)
            await self.broadcast_decision(value)
            return True
        
        return False
```

### 3.2 Raft算法

Raft是更易于理解的共识算法，将问题分解为领导选举、日志复制和安全性。

```go
// Raft核心实现
package raft

import (
	"math/rand"
	"sync"
	"time"
)

type RaftState int

const (
	Follower  RaftState = iota
	Candidate
	Leader
)

type RaftNode struct {
	mu          sync.Mutex
	id          int
	peers       []*RaftPeer
	currentTerm int
	votedFor    int
	log         []LogEntry
	state       RaftState
	
	// Volatile state on all servers
	commitIndex int
	lastApplied int
	
	// Volatile state on leaders
	nextIndex   []int
	matchIndex  []int
	
	// Election timeout
	electionTimer *time.Timer
	
	// Channel for state changes
	stateChange chan RaftState
}

type LogEntry struct {
	Term    int
	Command interface{}
}

func NewRaftNode(id int, peers []*RaftPeer) *RaftNode {
	node := &RaftNode{
		id:          id,
		peers:       peers,
		currentTerm: 0,
		votedFor:    -1,
		log:         make([]LogEntry, 0),
		state:       Follower,
		commitIndex: 0,
		lastApplied: 0,
		stateChange: make(chan RaftState, 10),
	}
	
	go node.run()
	return node
}

func (r *RaftNode) run() {
	for {
		switch r.state {
		case Follower:
			r.runFollower()
		case Candidate:
			r.runCandidate()
		case Leader:
			r.runLeader()
		}
	}
}

func (r *RaftNode) runFollower() {
	// 重置选举超时
	timeout := time.Duration(150+rand.Intn(150)) * time.Millisecond
	r.electionTimer = time.NewTimer(timeout)
	
	select {
	case <-r.electionTimer.C:
		// 选举超时，成为候选人
		r.mu.Lock()
		r.state = Candidate
		r.currentTerm++
		r.votedFor = r.id
		r.mu.Unlock()
		
	case <-r.stateChange:
		return
	}
}

func (r *RaftNode) runCandidate() {
	// 请求投票
	votes := 1 // 投票给自己
	
	for _, peer := range r.peers {
		go func(p *RaftPeer) {
			args := &RequestVoteArgs{
				Term:         r.currentTerm,
				CandidateId:  r.id,
				LastLogIndex: len(r.log) - 1,
				LastLogTerm:  r.lastLogTerm(),
			}
			
			reply, err := p.RequestVote(args)
			if err != nil {
				return
			}
			
			r.mu.Lock()
			defer r.mu.Unlock()
			
			if reply.Term > r.currentTerm {
				// 发现更高任期，转为追随者
				r.becomeFollower(reply.Term)
				return
			}
			
			if reply.VoteGranted {
				votes++
				if votes > len(r.peers)/2+1 {
					// 赢得选举，成为领导
					r.becomeLeader()
				}
			}
		}(peer)
	}
	
	// 等待选举结果或超时
	select {
	case <-r.stateChange:
		return
	case <-time.After(time.Duration(150+rand.Intn(150)) * time.Millisecond):
		// 选举超时，重新开始选举
		r.mu.Lock()
		r.currentTerm++
		r.votedFor = r.id
		r.mu.Unlock()
	}
}

func (r *RaftNode) runLeader() {
	// 初始化领导状态
	r.mu.Lock()
	for i := 0; i < len(r.peers); i++ {
		r.nextIndex[i] = len(r.log)
		r.matchIndex[i] = 0
	}
	r.mu.Unlock()
	
	// 定期发送心跳
	ticker := time.NewTicker(50 * time.Millisecond)
	defer ticker.Stop()
	
	for {
		select {
		case <-ticker.C:
			r.sendHeartbeats()
		case <-r.stateChange:
			return
		}
	}
}

func (r *RaftNode) sendHeartbeats() {
	for i, peer := range r.peers {
		go func(idx int, p *RaftPeer) {
			r.mu.Lock()
			args := &AppendEntriesArgs{
				Term:         r.currentTerm,
				LeaderId:     r.id,
				PrevLogIndex: r.nextIndex[idx] - 1,
				PrevLogTerm:  r.prevLogTerm(idx),
				Entries:      r.log[r.nextIndex[idx]:],
				LeaderCommit: r.commitIndex,
			}
			r.mu.Unlock()
			
			reply, err := p.AppendEntries(args)
			if err != nil {
				return
			}
			
			r.mu.Lock()
			defer r.mu.Unlock()
			
			if reply.Term > r.currentTerm {
				r.becomeFollower(reply.Term)
				return
			}
			
			if reply.Success {
				r.nextIndex[idx] = args.PrevLogIndex + len(args.Entries) + 1
				r.matchIndex[idx] = r.nextIndex[idx] - 1
			} else {
				r.nextIndex[idx] = max(1, r.nextIndex[idx]-1)
			}
			
			// 更新commitIndex
			r.updateCommitIndex()
		}(i, peer)
	}
}

func (r *RaftNode) updateCommitIndex() {
	for n := len(r.log) - 1; n > r.commitIndex; n-- {
		if r.log[n].Term != r.currentTerm {
			continue
		}
		
		count := 1 // 包括自己
		for _, match := range r.matchIndex {
			if match >= n {
				count++
			}
		}
		
		if count > len(r.peers)/2+1 {
			r.commitIndex = n
			r.applyStateMachine()
			break
		}
	}
}
```

### 3.3 ZAB算法

ZooKeeper Atomic Broadcast（ZAB）是ZooKeeper使用的共识协议。

```java
// ZAB协议核心实现
public class ZabProtocol {
    private final int nodeId;
    private final List<ZabPeer> peers;
    private final List<Proposal> proposals;
    
    private State state = State.FOLLOWING;
    private long currentEpoch = 0;
    private long lastZxid = 0;
    
    enum State {
        LOOKING, FOLLOWING, LEADING
    }
    
    // 领导选举
    public void electLeader() {
        state = State.LOOKING;
        
        // 1. 发送选举通知
        Map<Integer, Vote> votes = new HashMap<>();
        votes.put(nodeId, new Vote(nodeId, currentEpoch, lastZxid));
        
        // 2. 收集选票
        for (ZabPeer peer : peers) {
            try {
                Vote vote = peer.lookForLeader(new Vote(nodeId, currentEpoch, lastZxid));
                votes.put(peer.getId(), vote);
            } catch (Exception e) {
                // 忽略不可达节点
            }
        }
        
        // 3. 选择领导者（基于epoch和zxid）
        Vote leaderVote = selectLeader(votes);
        
        if (leaderVote.getNodeId() == nodeId) {
            becomeLeader();
        } else {
            becomeFollower(leaderVote.getNodeId());
        }
    }
    
    // 原子广播
    public void broadcast(Proposal proposal) {
        if (state != State.LEADING) {
            throw new IllegalStateException("Only leader can broadcast");
        }
        
        // 1. 分配zxid
        proposal.setZxid(generateZxid());
        
        // 2. 发送PROPOSAL
        for (ZabPeer peer : peers) {
            peer.sendProposal(proposal);
        }
        
        // 3. 等待ACK
        int acks = 1; // 自己的ACK
        for (ZabPeer peer : peers) {
            try {
                if (peer.waitForAck(proposal.getZxid())) {
                    acks++;
                }
            } catch (TimeoutException e) {
                // 超时处理
            }
        }
        
        // 4. 如果达到多数派，发送COMMIT
        if (acks > peers.size() / 2) {
            for (ZabPeer peer : peers) {
                peer.sendCommit(proposal.getZxid());
            }
            
            // 5. 应用到本地状态机
            applyProposal(proposal);
        }
    }
    
    // 恢复阶段
    public void recover() {
        // 1. 从领导者获取缺失的事务
        Proposal lastProposal = getLastProposal();
        
        for (ZabPeer leader : getLeaders()) {
            try {
                List<Proposal> missing = leader.getMissingProposals(
                    lastProposal.getZxid());
                
                // 2. 应用缺失的事务
                for (Proposal proposal : missing) {
                    applyProposal(proposal);
                }
                
                break;
            } catch (Exception e) {
                continue;
            }
        }
    }
    
    private long generateZxid() {
        // zxid = epoch + counter
        return (currentEpoch << 32) | (++lastZxid);
    }
}
```

```java
// 2PC协调者实现
public class TwoPhaseCommitCoordinator {
    private final TransactionManager tm;
    private final Map<String, ResourceManager> resourceManagers;
    
    public boolean commit(TransactionId txId) {
        // Phase 1: Prepare
        Map<String, Boolean> prepareResults = new HashMap<>();
        
        for (Map.Entry<String, ResourceManager> entry : resourceManagers.entrySet()) {
            try {
                boolean prepared = entry.getValue().prepare(txId);
                prepareResults.put(entry.getKey(), prepared);
            } catch (Exception e) {
                prepareResults.put(entry.getKey(), false);
            }
        }
        
        // 检查是否所有参与者都准备好了
        boolean allPrepared = prepareResults.values().stream()
            .allMatch(Boolean::booleanValue);
        
        if (allPrepared) {
            // Phase 2: Commit
            for (ResourceManager rm : resourceManagers.values()) {
                try {
                    rm.commit(txId);
                } catch (Exception e) {
                    // 记录日志，需要人工干预
                    logCommitFailure(txId, e);
                }
            }
            
            // 记录提交日志
            tm.logCommit(txId);
            return true;
        } else {
            // Phase 2: Rollback
            for (ResourceManager rm : resourceManagers.values()) {
                try {
                    rm.rollback(txId);
                } catch (Exception e) {
                    logRollbackFailure(txId, e);
                }
            }
            
            tm.logAbort(txId);
            return false;
        }
    }
}

public class TwoPhaseCommitParticipant {
    private final String resourceId;
    private final LocalTransactionManager ltm;

        try {
            // 1. 执行本地事务（但不提交）
            LocalTransaction localTx = ltm.beginTransaction();
            localTx.prepare(); // 写入prepare日志
            
            // 2. 获取资源锁
            lockResources(txId);
            
            // 3. 记录prepare日志（用于恢复）
            writePrepareLog(txId, localTx);
            
            return true;
        } catch (Exception e) {
            return false;
        }
    }
    
    public void commit(TransactionId txId) {
        // 1. 提交本地事务
        LocalTransaction localTx = ltm.getTransaction(txId);
        localTx.commit();
        
        // 2. 释放资源锁
        releaseResources(txId);
        
        // 3. 清理日志
        cleanupLogs(txId);
    }
    
    public void rollback(TransactionId txId) {
        // 1. 回滚本地事务
        LocalTransaction localTx = ltm.getTransaction(txId);
        localTx.rollback();
        
        // 2. 释放资源锁
        releaseResources(txId);
        
        // 3. 清理日志
        cleanupLogs(txId);
    }
```

### 4.2 三阶段提交（3PC）

```python
# 3PC实现
class ThreePhaseCoordinator:
    def __init__(self, participants):
        self.participants = participants
        
    def execute_transaction(self, transaction):
        # Phase 1: CanCommit?
        can_commit_responses = []
        for p in self.participants:
            try:
                response = p.can_commit(transaction)
                can_commit_responses.append(response)
            except Exception as e:
                can_commit_responses.append(False)
        
        # 检查是否所有参与者都能提交
        if not all(can_commit_responses):
            self.abort(transaction)
            return False
        
        # Phase 2: PreCommit
        pre_commit_responses = []
        for p in self.participants:
            try:
                response = p.pre_commit(transaction)
                pre_commit_responses.append(response)
            except Exception as e:
                pre_commit_responses.append(False)
        
        if not all(pre_commit_responses):
            self.abort(transaction)
            return False
        
        # Phase 3: DoCommit
        for p in self.participants:
            try:
                p.do_commit(transaction)
            except Exception as e:
                # 记录日志但继续
                self.log_commit_failure(transaction, p, e)
        
        return True

class ThreePhaseParticipant:
    def can_commit(self, transaction):
        # 检查是否能够提交（但不实际执行）
        try:
            # 验证资源
            if not self.validate_resources(transaction):
                return False
            
            # 检查约束
            if not self.check_constraints(transaction):
                return False
            
            return True
        except Exception as e:
            return False
    
    def pre_commit(self, transaction):
        try:
            # 准备资源
            self.prepare_resources(transaction)
            
            # 写入准备日志
            self.write_prepare_log(transaction)
            
            return True
        except Exception as e:
            return False
    
    def do_commit(self, transaction):
        try:
            # 实际执行事务
            self.execute_transaction(transaction)
            
            # 写入提交日志
            self.write_commit_log(transaction)
            
            # 释放资源
            self.release_resources(transaction)
            
            return True
        except Exception as e:
            # 3PC的优势：即使失败也能继续
            self.log_failure(transaction, e)
            raise
```

```go
// TCC模式实现
package tcc

import (
	"context"
	"fmt"
	"sync"
)

type TCCCoordinator struct {
	resources map[string]TCCResource
	mu        sync.Mutex
}

type TCCResource interface {
	Try(ctx context.Context, txID string) error
	Confirm(ctx context.Context, txID string) error
	Cancel(ctx context.Context, txID string) error
}

type TCCTransaction struct {
	ID        string
	Resources []string
	Tried     map[string]bool
}

func (c *TCCCoordinator) ExecuteTransaction(ctx context.Context, txID string, resources []string) error {
	tx := &TCCTransaction{
		ID:        txID,
		Resources: resources,
		Tried:     make(map[string]bool),
	}
	
	// Phase 1: Try
	if err := c.tryPhase(ctx, tx); err != nil {
		c.cancelPhase(ctx, tx)
		return err
	}
	
	// Phase 2: Confirm
	if err := c.confirmPhase(ctx, tx); err != nil {
		return err
	}
	
	return nil
}

func (c *TCCCoordinator) tryPhase(ctx context.Context, tx *TCCTransaction) error {
	for _, resID := range tx.Resources {
		resource, ok := c.resources[resID]
		if !ok {
			return fmt.Errorf("resource %s not found", resID)
		}
		
		// 调用Try操作
		if err := resource.Try(ctx, tx.ID); err != nil {
			// Try失败，取消所有已Try的资源
			return err
		}
		
		tx.Tried[resID] = true
	}
	
	return nil
}

func (c *TCCCoordinator) confirmPhase(ctx context.Context, tx *TCCTransaction) error {
	for _, resID := range tx.Resources {
		if !tx.Tried[resID] {
			continue
		}
		
		resource := c.resources[resID]
		
		if err := resource.Confirm(ctx, tx.ID); err != nil {
			// Confirm失败需要重试或记录
			c.logConfirmFailure(tx.ID, resID, err)
		}
	}
	
	return nil
}

func (c *TCCCoordinator) cancelPhase(ctx context.Context, tx *TCCTransaction) {
	for resID, tried := range tx.Tried {
		if !tried {
			continue
		}
		
		resource := c.resources[resID]
		
		if err := resource.Cancel(ctx, tx.ID); err != nil {
			// Cancel失败需要重试或记录
			c.logCancelFailure(tx.ID, resID, err)
		}
	}
}
```
