## 1. 云计算基础：理解服务模型

### IaaS、PaaS、SaaS、FaaS的本质区别

**IaaS（基础设施即服务）** 提供虚拟化的计算资源作为服务，用户需要管理操作系统及以上所有层。就像租用毛坯房，租客需要自己装修和布置家具。

```python
# 模拟IaaS环境配置示例（使用AWS EC2）
import boto3

def create_ec2_instance():
    ec2 = boto3.resource('ec2')
    
    # 这模拟了IaaS的灵活性：用户需要选择OS、配置安全组、管理存储
    instances = ec2.create_instances(
        ImageId='ami-0abcdef1234567890',  # 用户需要自己选择和维护镜像
        InstanceType='t3.micro',
        MinCount=1,
        MaxCount=1,
        SecurityGroupIds=['sg-12345678'],
        SubnetId='subnet-12345678',
        UserData='''#!/bin/bash
        # 用户需要自己安装和配置所有软件
        yum update -y
        yum install httpd -y
        systemctl start httpd
        systemctl enable httpd'''
    )
    return instances[0].id
```

**PaaS（平台即服务）** 提供完整的开发和部署环境，用户专注于代码开发。类似精装公寓，租客只需携带个人物品入住。

```python
# 使用AWS Elastic Beanstalk（PaaS服务）的示例
import boto3
import zipfile
import io

def deploy_to_beanstalk():
    eb = boto3.client('elasticbeanstalk')
    
    # 创建应用包（用户只需提供代码，无需管理服务器）
    zip_buffer = io.BytesIO()
    with zipfile.ZipFile(zip_buffer, 'w', zipfile.ZIP_DEFLATED) as zip_file:
        zip_file.writestr('application.py', '''
from flask import Flask
app = Flask(__name__)

@app.route("/")
def hello():
    return "Hello from PaaS!"

if __name__ == "__main__":
    app.run(host="0.0.0.0")
        ''')
        zip_file.writestr('requirements.txt', 'flask==2.0.1')
    
    # 部署到Elastic Beanstalk
    eb.create_application_version(
        ApplicationName='my-flask-app',
        VersionLabel='v1',
        SourceBundle={
            'S3Bucket': 'my-app-deployments',
            'S3Key': 'app.zip'
        }
    )
    
    eb.update_environment(
        EnvironmentName='my-env',
        VersionLabel='v1'
    )
```

**SaaS（软件即服务）** 提供完整的软件解决方案，用户通过网络访问。就像使用外卖平台，无需关心厨房和配送细节。

**FaaS（函数即服务）** 是无服务器计算的核心，用户只关心函数逻辑。如同使用共享厨房按次付费烹饪。

```python
# AWS Lambda函数示例（FaaS）
def lambda_handler(event, context):
    """
    用户只需编写这个函数，无需管理服务器
    AWS Lambda会自动处理：
    - 服务器配置和管理
    - 自动扩缩容
    - 高可用性
    - 操作系统维护
    """
    import json
    
    # 处理API Gateway请求
    if 'httpMethod' in event:
        body = json.loads(event['body']) if event.get('body') else {}
        
        return {
            'statusCode': 200,
            'headers': {'Content-Type': 'application/json'},
            'body': json.dumps({
                'message': 'Hello from FaaS!',
                'function': context.function_name,
                'event': event
            })
        }
    
    # 处理其他事件源（S3、SQS等）
    else:
        print(f"Event received: {event}")
        return {
            'statusCode': 200,
            'body': 'Event processed'
        }
```

### 服务模型对比分析

| 维度 | IaaS | PaaS | SaaS | FaaS |
|------|------|------|------|------|
| **管理责任** | OS、中间件、应用 | 仅应用代码 | 仅数据和使用配置 | 仅函数逻辑 |
| **扩展性** | 手动或脚本 | 自动（平台管理） | 完全自动 | 事件驱动自动扩展 |
| **成本模型** | 按资源预留计费 | 按平台使用计费 | 按用户/订阅计费 | 按执行次数和时长 |
| **典型服务** | AWS EC2, 阿里云ECS | AWS Elastic Beanstalk, Heroku | Salesforce, Slack | AWS Lambda, 阿里云函数计算 |
| **适用场景** | 需要完全控制的场景 | 快速应用开发和部署 | 标准化业务软件 | 事件驱动、微服务 |

### 2.2 S3（Simple Storage Service）

S3是对象存储服务，提供99.999999999%的持久性和几乎无限的存储容量。

```python
# S3高级功能示例：静态网站托管 + CloudFront + 版本控制
import boto3
import json
from botocore.exceptions import ClientError

def setup_s3_website():
    s3 = boto3.client('s3')
    bucket_name = 'my-unique-bucket-name-2024'
    
    # 创建存储桶
    try:
        s3.create_bucket(Bucket=bucket_name)
    except ClientError as e:
        if e.response['Error']['Code'] == 'BucketAlreadyOwnedByYou':
            print("Bucket already exists")
        else:
            raise
    
    # 启用版本控制
    s3.put_bucket_versioning(
        Bucket=bucket_name,
        VersioningConfiguration={
            'Status': 'Enabled'
        }
    )
    
    # 设置存储桶策略（允许公共读取）
    bucket_policy = {
        "Version": "2012-10-17",
        "Statement": [
            {
                "Sid": "PublicReadGetObject",
                "Effect": "Allow",
                "Principal": "*",
                "Action": "s3:GetObject",
                "Resource": f"arn:aws:s3:::{bucket_name}/*"
            }
        ]
    }
    
    s3.put_bucket_policy(
        Bucket=bucket_name,
        Policy=json.dumps(bucket_policy)
    )
    
    # 配置静态网站托管
    s3.put_bucket_website(
        Bucket=bucket_name,
        WebsiteConfiguration={
            'IndexDocument': {'Suffix': 'index.html'},
            'ErrorDocument': {'Key': 'error.html'},
            'RoutingRules': [
                {
                    'Condition': {'KeyPrefixEquals': 'docs/'},
                    'Redirect': {'ReplaceKeyPrefixWith': 'documents/'}
                }
            ]
        }
    )
    
    # 上传网站文件
    s3.put_object(
        Bucket=bucket_name,
        Key='index.html',
        Body='''
<!DOCTYPE html>
<html>
<head><title>S3 Static Website</title></head>
<body>
<h1>Welcome to S3 Static Website</h1>
<p>Hosted on AWS S3</p>
</body>
</html>
        ''',
        ContentType='text/html'
    )
    
    # 设置生命周期策略（成本优化）
    lifecycle_config = {
        'Rules': [
            {

                    {
                        'Days': 90,
                        'StorageClass': 'GLACIER'
                    }

                'NoncurrentVersionTransitions': [
                    {
                        'NoncurrentDays': 30,

            }

    }

        Bucket=bucket_name,

    return f"http://{bucket_name}.s3-website-us-east-1.amazonaws.com"
```

```python
            'Access-Control-Allow-Origin': '*'
```

```python
        # 设置首页和404页面
        website_config = oss2.models.WebsiteConfiguration(
            index_document='index.html',
            error_document='error.html'
        )
        
        # 阿里云特有功能：镜像回源
        website_config.set_mirror_rules([
            oss2.models.MirrorRule(
                source_url='https://source.example.com',
                prefix='static/',
                pass_query_string=True,
                follow_redirect=True
            )
        ])
        
        self.bucket.put_bucket_website(website_config)
    
    def setup_cors(self):
        """配置跨域资源共享"""
        rule = CorsRule(
            allowed_origins=['*'],
            allowed_methods=['GET', 'POST', 'PUT', 'DELETE', 'HEAD'],
            allowed_headers=['*'],
            expose_headers=['x-oss-request-id', 'ETag'],
            max_age_seconds=3600
        )
        
        bucket_cors = BucketCors()
        bucket_cors.rules = [rule]
        self.bucket.put_bucket_cors(bucket_cors)
    
    def enable_encryption(self):
        """启用服务器端加密（阿里云KMS加密）"""
        # 阿里云支持KMS托管密钥加密
```
