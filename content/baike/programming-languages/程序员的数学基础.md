```python
import heapq
from collections import defaultdict, deque

class Graph:
    """图的通用实现"""
    
    def __init__(self, directed=False):
        self.graph = defaultdict(list)
        self.directed = directed
        self.vertices = set()
    
    def add_edge(self, u, v, weight=1):
        """添加边"""
        self.graph[u].append((v, weight))
        self.vertices.add(u)
        self.vertices.add(v)
        
        if not self.directed:
            self.graph[v].append((u, weight))
    
    def dijkstra(self, start):
        """Dijkstra最短路径算法"""
        distances = {v: float('infinity') for v in self.vertices}
        distances[start] = 0
        pq = [(0, start)]
        
        while pq:
            current_distance, current_vertex = heapq.heappop(pq)
            
            # 如果找到更短路径，跳过
            if current_distance > distances[current_vertex]:
                continue
            
            for neighbor, weight in self.graph[current_vertex]:
                distance = current_distance + weight
                
                # 如果找到更短路径
                if distance < distances[neighbor]:
                    distances[neighbor] = distance
                    heapq.heappush(pq, (distance, neighbor))
        
        return distances
    
    def bfs(self, start):
        """广度优先搜索"""
        visited = set()
        queue = deque([start])
        traversal_order = []
        
        while queue:
            vertex = queue.popleft()
            if vertex not in visited:
                visited.add(vertex)
                traversal_order.append(vertex)
                
                for neighbor, _ in self.graph[vertex]:
                    if neighbor not in visited:
                        queue.append(neighbor)
        
        return traversal_order
    
    def has_cycle(self):
        """检测图中是否存在环（仅适用于有向图）"""
        if not self.directed:
            raise ValueError("Cycle detection only works for directed graphs")
        
        visited = set()
        recursion_stack = set()
        
        def dfs(vertex):
            visited.add(vertex)
            recursion_stack.add(vertex)
            
            for neighbor, _ in self.graph[vertex]:
                if neighbor not in visited:
                    if dfs(neighbor):
                        return True
                elif neighbor in recursion_stack:
                    return True
            
            recursion_stack.remove(vertex)
            return False
        
        for vertex in self.vertices:
            if vertex not in visited:
                if dfs(vertex):
                    return True
        return False

# 实际应用：社交网络分析
class SocialNetworkAnalyzer:

        # 按度排序

        return [node for node, _ in sorted_nodes[:k]]

        direct_friends = {v for v, _ in self.network.graph[user]}
```

```python
@lru_cache(maxsize=None)
def combination_dp(n, r):
    """使用动态规划计算组合数，避免重复计算"""
    if r == 0 or r == n:
        return 1
    return combination_dp(n-1, r-1) + combination_dp(n-1, r)

# 容斥原理示例：求1到n中能被a或b整除的数的个数
def count_divisible(n, a, b):
    """使用容斥原理计算"""
    count_a = n // a
    count_b = n // b
    count_ab = n // math.lcm(a, b)  # 能同时被a和b整除的数
    
    # 容斥原理：|A ∪ B| = |A| + |B| - |A ∩ B|
    return count_a + count_b - count_ab

# 生成排列的递归实现
def generate_permutations(arr):
    """生成所有排列"""
    result = []
    
    def backtrack(start):
        if start == len(arr):
            result.append(arr[:])
            return
        
        for i in range(start, len(arr)):
            arr[start], arr[i] = arr[i], arr[start]  # 交换
            backtrack(start + 1)
            arr[start], arr[i] = arr[i], arr[start]  # 回溯
    
    backtrack(0)
    return result

class PasswordStrengthCalculator:
    """密码强度计算器"""
    
    def calculate_entropy(self, password):
        """计算密码的熵（信息量）"""
        if not password:
            return 0
        
        # 统计字符集大小
        char_sets = {
            'lowercase': set('abcdefghijklmnopqrstuvwxyz'),
            'uppercase': set('ABCDEFGHIJKLMNOPQRSTUVWXYZ'),
            'digits': set('0123456789'),
            'special': set('!@#$%^&*()_+-=[]{}|;:,.<>?')
        }
        
        used_chars = set(password)
        charset_size = 0
        
        for chars in char_sets.values():
            if used_chars & chars:
                charset_size += len(chars)
        
        # 熵 = 长度 * log2(字符集大小)
        if charset_size == 0:
            return 0
        
        entropy = len(password) * math.log2(charset_size)
        return entropy
    
    def estimate_crack_time(self, entropy):
        """估算破解时间（假设每秒尝试1亿次）"""
        # 2^熵次尝试
        total_attempts = 2 ** entropy
        attempts_per_second = 1e8
        
        seconds = total_attempts / attempts_per_second
        
        # 转换为易读格式
        if seconds < 60:
            return f"{seconds:.2f}秒"
        elif seconds < 3600:
            return f"{seconds/60:.2f}分钟"
        elif seconds < 86400:
            return f"{seconds/3600:.2f}小时"
        elif seconds < 31536000:
```

```python
class LogicalDatabase:

class PropositionalSolver:
    """命题逻辑求解器"""
    
    def evaluate(self, formula, assignment):
        """根据变量赋值评估逻辑公式"""
        # 简单的递归下降解析器
        if isinstance(formula, str):
            return assignment.get(formula, False)
        
        operator = formula[0]
        
        if operator == 'NOT':
            return not self.evaluate(formula[1], assignment)
        elif operator == 'AND':
            left = self.evaluate(formula[1], assignment)
            right = self.evaluate(formula[2], assignment)
            return left and right
        elif operator == 'OR':
            left = self.evaluate(formula[1], assignment)
            right = self.evaluate(formula[2], assignment)
            return left or right
        elif operator == 'IMPLIES':
            antecedent = self.evaluate(formula[1], assignment)
            consequent = self.evaluate(formula[2], assignment)
            return (not antecedent) or consequent
        elif operator == 'IFF':
            left = self.evaluate(formula[1], assignment)
            right = self.evaluate(formula[2], assignment)
            return left == right
        
        return False
    
    def find_models(self, formula, variables):
        """找出使公式为真的所有变量赋值（模型）"""
        models = []
        
        # 生成所有可能的真值赋值
        from itertools import product
        for values in product([True, False], repeat=len(variables)):
            assignment = dict(zip(variables, values))
            if self.evaluate(formula, assignment):
                models.append(assignment)
        
        return models
    
    def is_tautology(self, formula, variables):
        """检查公式是否为重言式（永真式）"""
        models = self.find_models(formula, variables)
        total_models = 2 ** len(variables)
        return len(models) == total_models

class TypeSystem:
    """简单的类型系统，基于谓词逻辑"""
    
    def __init__(self):
        self.types = {}
        self.subtype_relations = {}
    
    def add_type(self, type_name, predicates):
        """添加类型及其谓词"""
        self.types[type_name] = predicates
    
    def add_subtype(self, child, parent):
        """添加子类型关系"""
        if child not in self.subtype_relations:
            self.subtype_relations[child] = set()
        self.subtype_relations[child].add(parent)
    
    def is_instance(self, obj, type_name):
        """检查对象是否为给定类型的实例"""
        if type_name not in self.types:
            return False
        
        predicates = self.types[type_name]
        return all(pred(obj) for pred in predicates)
    
    def is_subtype(self, child_type, parent_type):
        """检查子类型关系"""
        if child_type == parent_type:
            return True
        
        if child_type not in self.subtype_relations:
            return False
        
        # 递归检查所有父类型
        for direct_parent in self.subtype_relations[child_type]:
            if self.is_subtype(direct_parent, parent_type):
                return True
        
        return False
```

```python
import numpy as np

class Vector:
    """自定义向量类，演示向量运算"""
    
    def __init__(self, components):
        self.components = np.array(components, dtype=float)
        self.dimension = len(components)
    
    def __add__(self, other):
        """向量加法"""
        if self.dimension != other.dimension:
            raise ValueError("Vectors must have same dimension")
        return Vector(self.components + other.components)
    
    def __sub__(self, other):
        """向量减法"""
        if self.dimension != other.dimension:
            raise ValueError("Vectors must have same dimension")
        return Vector(self.components - other.components)
    
    def __mul__(self, scalar):
        """标量乘法"""
        return Vector(self.components * scalar)
    
    def __rmul__(self, scalar):
        """反向标量乘法"""
        return self * scalar
    
    def dot(self, other):
        """点积"""
        if self.dimension != other.dimension:
            raise ValueError("Vectors must have same dimension")
        return np.dot(self.components, other.components)
    
    def cross(self, other):
        """叉积（仅适用于3D向量）"""
        if self.dimension != 3 or other.dimension != 3:
            raise ValueError("Cross product only defined for 3D vectors")
        return Vector(np.cross(self.components, other.components))
    
    def norm(self, p=2):
        """向量范数"""
        return np.linalg.norm(self.components, ord=p)
    
    def normalize(self):
        """单位化向量"""
        norm = self.norm()
        if norm == 0:
            return Vector(self.components)
        return Vector(self.components / norm)
    
    def angle_with(self, other):
        """计算与另一向量的夹角（弧度）"""
        cos_theta = self.dot(other) / (self.norm() * other.norm())
        # 数值稳定性：确保在[-1, 1]范围内
        cos_theta = np.clip(cos_theta, -1.0, 1.0)
        return np.arccos(cos_theta)
    
    def projection_onto(self, other):
        """计算在另一向量上的投影"""
        other_norm_sq = other.dot(other)
        if other_norm_sq == 0:
            return Vector(np.zeros(self.dimension))
        
        scalar = self.dot(other) / other_norm_sq
        return other * scalar

class PhysicsEngine:

            obj['velocity'] += acceleration * dt

    def check_collision(self, obj1, obj2, radius1=1.0, radius2=1.0):
        """检查两个圆形对象是否碰撞"""
        distance = np.linalg.norm(obj1['position'] - obj2['position'])
        return distance <= (radius1 + radius2)
    
    def resolve_collision(self, obj1, obj2, restitution=0.8):
        """解决碰撞（弹性碰撞）"""
        # 计算相对速度
        relative_velocity = obj1['velocity'] - obj2['velocity']
        
        # 碰撞法线方向
        normal = obj2['position'] - obj1['position']
        normal_norm = np.linalg.norm(normal)
        if normal_norm == 0:
            return
        
        normal = normal / normal_norm
        
        # 相对速度在法线方向的分量
        relative_velocity_normal = np.dot(relative_velocity, normal)
        
        # 如果对象正在远离，则不处理
        if relative_velocity_normal > 0:
            return
        
        # 计算冲量（考虑质量）
        total_mass = obj1['mass'] + obj2['mass']
        impulse = -(1 + restitution) * relative_velocity_normal
        impulse /= (1/obj1['mass'] + 1/obj2['mass'])
        
        # 应用冲量
        impulse_vector = impulse * normal
        obj1['velocity'] += impulse_vector / obj1['mass']
        obj2['velocity'] -= impulse_vector / obj2['mass']

# 向量在机器学习中的应用：相似性计算
class SimilarityCalculator:
    """基于向量的相似性计算"""
    
    @staticmethod
    def cosine_similarity(vec_a, vec_b):
        """余弦相似度"""
        dot_product = np.dot(vec_a, vec_b)
        norm_a = np.linalg.norm(vec_a)
        norm_b = np.linalg.norm(vec_b)
        
        if norm_a == 0 or norm_b == 0:
            return 0
        
        return dot_product / (norm_a * norm_b)
    
    @staticmethod
    def euclidean_distance(vec_a, vec_b):
        """欧几里得距离"""
        return np.linalg.norm(vec_a - vec_b)
    
    @staticmethod
    def manhattan_distance(vec_a, vec_b):
        """曼哈顿距离"""
        return np.sum(np.abs(vec_a - vec_b))
    
    @staticmethod
    def jaccard_similarity(set_a, set_b):
        """杰卡德相似度（集合相似度）"""
        intersection = len(set_a & set_b)
        union = len(set_a | set_b)
        
        if union == 0:
            return 0
        
        return intersection / union

class DocumentVectorizer:
    """将文本文档转换为向量"""
    
    def __init__(self):
        self.vocabulary = {}
        self.idf = {}
    
    def build_vocabulary(self, documents):
        """构建词汇表"""
        word_counts = {}
        doc_counts = {}
        
        for doc in documents:
            words = self.tokenize(doc)
            unique_words = set(words)
            
            for word in words:
                word_counts[word] = word_counts.get(word, 0) + 1
            
            for word in unique_words:
                doc_counts[word] = doc_counts.get(word, 0) + 1
        
        # 构建词汇表映射
        self.vocabulary = {word: idx for idx, word in enumerate(word_counts.keys())}
        
        # 计算IDF
        n_docs = len(documents)
        self.idf = {}
        for word, count in doc_counts.items():
            self.idf[word] = np.log(n_docs / (1 + count))

        """简单的分词"""
        import re
        return re.findall(r'\w+', text.lower())
    
    def document_to_vector(self, document):
        """将文档转换为TF-IDF向量"""
        words = self.tokenize(document)
        word_counts = {}
        
        for word in words:
            word_counts[word] = word_counts.get(word, 0) + 1
        
        vector = np.zeros(len(self.vocabulary))
        
        for word, count in word_counts.items():
            if word in self.vocabulary:
                idx = self.vocabulary[word]
                tf = count / len(words)  # 词频
                idf = self.idf.get(word, 0)
                vector[idx] = tf * idf
        
        return vector
```

```python
class PCA:
    """主成分分析实现"""
    
    def __init__(self, n_components=None):
        self.n_components = n_components
        self.components = None
        self.mean = None
        self.explained_variance = None
    
    def fit(self, X):
        """拟合PCA模型"""
        # 中心化数据
        self.mean = np.mean(X, axis=0)
        X_centered = X - self.mean
        
        # 计算协方差矩阵
        cov_matrix = np.cov(X_centered.T)
        
        # 计算特征值和特征向量
        eigenvalues, eigenvectors = np.linalg.eigh(cov_matrix)
        
        # 按特征值降序排序
        idx = np.argsort(eigenvalues)[::-1]
        eigenvalues = eigenvalues[idx]
        eigenvectors = eigenvectors[:, idx]
        
        # 选择前n_components个主成分
        if self.n_components is not None:
            self.components = eigenvectors[:, :self.n_components]
            self.explained_variance = eigenvalues[:self.n_components]
        else:
            self.components = eigenvectors
            self.explained_variance = eigenvalues
        
        return self
    
    def transform(self, X):
        """将数据投影到主成分空间"""
        X_centered = X - self.mean
        return np.dot(X_centered, self.components)
    
    def inverse_transform(self, X_transformed):
        """将投影数据转换回原始空间"""
        return np.dot(X_transformed, self.components.T) + self.mean
    
    def explained_variance_ratio(self):
        """解释方差比率"""
        total_variance = np.sum(self.explained_variance)
        return self.explained_variance / total_variance

class MatrixFactorization:
    """矩阵分解用于推荐系统"""
    
    def __init__(self, n_factors=10, learning_rate=0.01, regularization=0.02):
        self.n_factors = n_factors
        self.learning_rate = learning_rate
        self.regularization = regularization
        self.user_factors = None
        self.item_factors = None
    
    def fit(self, R, n_epochs=100):
        """
        训练矩阵分解模型
        R: 用户-物品评分矩阵
        """
        n_users, n_items = R.shape
        
        # 初始化因子矩阵
        self.user_factors = np.random.normal(0, 0.1, (n_users, self.n_factors))
        self.item_factors = np.random.normal(0, 0.1, (n_items, self.n_factors))
        
        # 训练过程
        for epoch in range(n_epochs):
            for i in range(n_users):
                for j in range(n_items):
                    if R[i, j] > 0:  # 只对有评分的项进行训练
                        # 计算预测误差
                        prediction = np.dot(self.user_factors[i], self.item_factors[j])
                        error = R[i, j] - prediction
                        
                        # 更新因子
                        self.user_factors[i] += self.learning_rate * (
                            error * self.item_factors[j] - 
                            self.regularization * self.user_factors[i]
                        )
                        self.item_factors[j] += self.learning_rate * (
                            error * self.user_factors[i] - 
                            self.regularization * self.item_factors[j]
                        )
        
        return self
    
    def predict(self, user_idx, item_idx):
        """预测评分"""
        return np.dot(self.user_factors[user_idx], self.item_factors[item_idx])
    
    def recommend(self, user_idx, n_recommendations=5):
        """为用户生成推荐"""
        # 计算所有物品的预测评分
        predictions = np.dot(self.user_factors[user_idx], self.item_factors.T)
        
        # 返回评分最高的物品索引
        return np.argsort(predictions)[::-1][:n_recommendations]
```

import numpy as np
from scipy import linalg

class SVDAnalyzer:
    """SVD分解分析器"""
    
    @staticmethod
    def svd_decomposition(A):
        """计算矩阵的SVD分解"""
        U, s, Vt = np.linalg.svd(A, full_matrices=False)
        return U, s, Vt
    
    @staticmethod
    def low_rank_approximation(A, rank):
        """低秩近似"""
        U, s, Vt = np.linalg.svd(A, full_matrices=False)
        
        # 保留前rank个奇异值
        U_k = U[:, :rank]
        s_k = s[:rank]
        Vt_k = Vt[:rank, :]
        
        # 重构近似矩阵
        return np.dot(U_k, np.dot(np.diag(s_k), Vt_k))
    
    @staticmethod
    def image_compression(image, rank):
        """使用SVD进行图像压缩"""
        # 分离颜色通道
        if len(image.shape) == 3:
            # 对每个通道分别进行SVD
            compressed_channels = []
            for channel in range(image.shape[2]):
                channel_data = image[:, :, channel]
                compressed_channel = SVDAnalyzer.low_rank_approximation(
                    channel_data, rank
                )
                compressed_channels.append(compressed_channel)
            
            # 合并通道
            compressed_image = np.stack(compressed_channels, axis=2)
        else:
            # 灰度图像
            compressed_image = SVDAnalyzer.low_rank_approximation(image, rank)
        
        return np.clip(compressed_image, 0, 255).astype(np.uint8)
    
    @staticmethod
    def compute_compression_ratio(original_shape, rank):
        """计算压缩比"""
        m, n = original_shape
        
        # 原始存储：m * n 个元素
        original_size = m * n
        
        # 压缩后存储：U_k (m*k), s_k (k), Vt_k (k*n)
        compressed_size = m * rank + rank + rank * n
        
        return original_size / compressed_size
