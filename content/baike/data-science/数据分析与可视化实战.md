### 1.1 NumPy - 数值计算基石

NumPy是Python科学计算的基础包，提供多维数组对象、派生对象（如掩码数组和矩阵）以及用于快速操作数组的例程。

**核心特性：**
- N维数组对象（ndarray）
- 广播功能函数
- 整合C/C++/Fortran代码的工具
- 线性代数、傅里叶变换和随机数生成

```python

def advanced_missing_value_treatment(df, strategy='auto'):
    """
    高级缺失值处理函数
    """
    df_processed = df.copy()
    
    # 数值列
    numeric_cols = df.select_dtypes(include=[np.number]).columns
    for col in numeric_cols:
        missing_ratio = df[col].isnull().mean()
        
        if missing_ratio < 0.05:  # 缺失比例低于5%，使用均值填充
            df_processed[col].fillna(df[col].mean(), inplace=True)
        elif missing_ratio < 0.2:  # 缺失比例5%-20%，使用中位数填充
            df_processed[col].fillna(df[col].median(), inplace=True)
        else:  # 缺失比例高于20%，标记为特殊值

    # 类别列

    for col in category_cols:
        df_processed[col].fillna('Unknown', inplace=True)
    
    # 日期列
    date_cols = df.select_dtypes(include=['datetime64']).columns
    for col in date_cols:
        df_processed[col].fillna(df[col].mode()[0], inplace=True)
```

```python
def detect_outliers_zscore(data, threshold=3):
    z_scores = np.abs(stats.zscore(data))
    return np.where(z_scores > threshold)[0]

z_outliers = detect_outliers_zscore(df_outliers['数值'])

# IQR方法
def detect_outliers_iqr(data, factor=1.5):
    Q1 = np.percentile(data, 25)
    Q3 = np.percentile(data, 75)
    IQR = Q3 - Q1
    lower_bound = Q1 - factor * IQR
    upper_bound = Q3 + factor * IQR
    return np.where((data < lower_bound) | (data > upper_bound))[0]

# 3. 异常值处理方法
def handle_outliers(df, column, method='iqr', factor=1.5):
    """
    处理异常值
    """
    df_processed = df.copy()
    data = df_processed[column]
    
    if method == 'iqr':
        Q1 = data.quantile(0.25)
        Q3 = data.quantile(0.75)
        IQR = Q3 - Q1
        lower_bound = Q1 - factor * IQR
        upper_bound = Q3 + factor * IQR
        
        # 方法1: 删除异常值
        df_processed = df_processed[
            (data >= lower_bound) & (data <= upper_bound)
        ]
        
        # 方法2: 截尾处理
        df_processed[column] = data.clip(lower=lower_bound, upper=upper_bound)
        
        # 方法3: 替换为边界值
        df_processed.loc[data < lower_bound, column] = lower_bound
        df_processed.loc[data > upper_bound, column] = upper_bound
    
    elif method == 'zscore':
        z_scores = np.abs(stats.zscore(data))
        df_processed = df_processed[z_scores < factor]
    
    elif method == 'percentile':
        lower_percentile = np.percentile(data, factor)
        upper_percentile = np.percentile(data, 100 - factor)
        df_processed[column] = data.clip(lower=lower_percentile, upper=upper_percentile)
    
    return df_processed

# 应用不同处理方法

print(f"Z-score处理后: {df_zscore.shape}")
print(f"百分位处理后: {df_percentile.shape}")
```

```python
from fuzzywuzzy import fuzz, process

def find_similar_duplicates(df, column, threshold=80):
    """
    使用模糊匹配查找相似记录
    """
    unique_values = df[column].unique()
    similar_pairs = []
    
    for i, val1 in enumerate(unique_values):
        for val2 in unique_values[i+1:]:
            similarity = fuzz.ratio(str(val1), str(val2))
            if similarity >= threshold:
                similar_pairs.append((val1, val2, similarity))
    
    return similar_pairs
```

```python
# 4. 高级类型转换函数
def smart_type_conversion(df):
    """
    智能类型转换
    """
    df_converted = df.copy()
    
    for col in df.columns:
        col_data = df_converted[col]
        
        # 尝试转换为数值类型
        if col_data.dtype == 'object':
            # 尝试转换为数值
            numeric_data = pd.to_numeric(col_data, errors='coerce')
            if not numeric_data.isna().all():
                df_converted[col] = numeric_data
                continue
            
            # 尝试转换为日期
            try:
                date_data = pd.to_datetime(col_data)
                if not date_data.isna().all():
                    df_converted[col] = date_data
                    continue
            except:
                pass
            
            # 检查是否为布尔值
            unique_vals = col_data.dropna().unique()
            if len(unique_vals) == 2:
                bool_mapping = {}
                for val in unique_vals:
                    if str(val).lower() in ['true', '是', 'yes', '1']:
                        bool_mapping[val] = True
                    elif str(val).lower() in ['false', '否', 'no', '0']:
                        bool_mapping[val] = False
                
                if len(bool_mapping) == 2:
                    df_converted[col] = col_data.map(bool_mapping)
                    continue
            
            # 转换为分类类型（如果唯一值较少）
            if len(unique_vals) < len(col_data) * 0.05:
                df_converted[col] = col_data.astype('category')
```

```python
        """基本信息分析"""

        print("数据集基本信息")
        print("=" * 60)
        
        print(f"数据形状: {self.df.shape}")
        print(f"行数: {self.df.shape[0]}")
        print(f"列数: {self.df.shape[1]}")
        
        print(f"\n数据类型统计:")
        print(self.df.dtypes.value_counts())
        
        print(f"\n缺失值统计:")
        missing = self.df.isnull().sum()
        missing_pct = missing / len(self.df) * 100
        missing_df = pd.DataFrame({
            '缺失数量': missing,
            '缺失比例(%)': missing_pct
        }).sort_values('缺失比例(%)', ascending=False)
        
        missing_df = missing_df[missing_df['缺失数量'] > 0]
        if len(missing_df) > 0:
            print(missing_df)
        else:
            print("无缺失值")
        
        print(f"\n重复行数量: {self.df.duplicated().sum()}")
        
        return missing_df
    
    def numeric_analysis(self):
        """数值变量分析"""
        if not self.numeric_cols:
            print("无数值变量")
            return
        
        print("\n" + "=" * 60)
        print("数值变量分析")
        print("=" * 60)
        
        # 描述性统计
        desc_stats = self.df[self.numeric_cols].describe()
        print("\n描述性统计:")
        print(desc_stats)
        
        # 分布分析
        n_cols = min(3, len(self.numeric_cols))
        n_rows = (len(self.numeric_cols) + n_cols - 1) // n_cols
        
        fig, axes = plt.subplots(n_rows, n_cols, figsize=(5*n_cols, 4*n_rows))
        if n_rows * n_cols == 1:
            axes = np.array([axes])
        axes = axes.flatten()
        
        for i, col in enumerate(self.numeric_cols):
            # 直方图
            axes[i].hist(self.df[col].dropna(), bins=30, edgecolor='black', alpha=0.7)
            axes[i].set_title(f'{col} 分布')
            axes[i].set_xlabel(col)
            axes[i].set_ylabel('频数')
            
            # 添加统计信息
            mean_val = self.df[col].mean()
            median_val = self.df[col].median()
            axes[i].axvline(mean_val, color='red', linestyle='--', label=f'均值: {mean_val:.2f}')
            axes[i].axvline(median_val, color='green', linestyle='--', label=f'中位数: {median_val:.2f}')
            axes[i].legend()
        
        # 隐藏空余的子图
        for j in range(i+1, len(axes)):
            axes[j].set_visible(False)
        
        plt.tight_layout()
        plt.show()
        
        # 偏度和峰度分析
        print("\n偏度和峰度:")
        skew_kurt = pd.DataFrame({
            '偏度': self.df[self.numeric_cols].skew(),
            '峰度': self.df[self.numeric_cols].kurtosis()
        })
        print(skew_kurt)
        
        return skew_kurt

            plt.ylabel('数量')
            plt.xticks(rotation=45)
            
            # 饼图（如果唯一值不多）
            plt.subplot(1, 2, 2)
            if len(value_counts) <= 10:
                value_counts.plot(kind='pie', autopct='%1.1f%%')
```
