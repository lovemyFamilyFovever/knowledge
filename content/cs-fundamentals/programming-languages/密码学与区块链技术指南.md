# 密码学与区块链技术指南

## 第一部分：密码学基础

密码学是区块链技术的基石，它为数字资产的安全传输、身份验证和数据完整性提供了理论基础。

### 1.1 对称加密

对称加密是最古老、最直接的加密方式，其核心思想是加密和解密使用相同的密钥。

**基本原理**：发送方使用密钥K将明文P加密为密文C，接收方使用相同的密钥K将密文C解密回明文P。数学表示为：
- 加密：E_K(P) = C
- 解密：D_K(C) = P

**主要特性**：
- **速度快**：相比非对称加密，计算开销小得多
- **密钥管理困难**：需要安全地共享和存储密钥
- **适合大数据量**：常用于加密文件、数据库和通信内容

**常见模式**：
- **ECB（电子密码本）**：每个块独立加密，相同明文产生相同密文，不安全
- **CBC（密码块链接）**：每个块与前一个密文块异或后再加密
- **GCM（伽罗瓦/计数器模式）**：提供认证加密，同时保证机密性和完整性

```python
# Python AES对称加密示例
from cryptography.hazmat.primitives.ciphers import Cipher, algorithms, modes
from cryptography.hazmat.backends import default_backend
import os

def aes_encrypt(plaintext, key):
    # 生成随机初始化向量
    iv = os.urandom(16)
    
    # 创建AES加密器（CBC模式）
    cipher = Cipher(algorithms.AES(key), modes.CBC(iv), backend=default_backend())
    encryptor = cipher.encryptor()
    
    # PKCS7填充
    padded_data = plaintext + (16 - len(plaintext) % 16) * chr(16 - len(plaintext) % 16)
    
    # 加密
    ciphertext = encryptor.update(padded_data.encode('utf-8')) + encryptor.finalize()
    
    return iv + ciphertext

def aes_decrypt(ciphertext, key):
    # 提取IV（前16字节）
    iv = ciphertext[:16]
    actual_ciphertext = ciphertext[16:]
    
    # 创建解密器
    cipher = Cipher(algorithms.AES(key), modes.CBC(iv), backend=default_backend())
    decryptor = cipher.decryptor()
    
    # 解密
    padded_plaintext = decryptor.update(actual_ciphertext) + decryptor.finalize()
    
    # 移除PKCS7填充
    padding_len = padded_plaintext[-1]
    plaintext = padded_plaintext[:-padding_len]
    
    return plaintext.decode('utf-8')

# 使用示例
key = os.urandom(32)  # AES-256
plaintext = "Hello, Blockchain World!"
encrypted = aes_encrypt(plaintext, key)
decrypted = aes_decrypt(encrypted, key)
print(f"原文: {plaintext}")
print(f"密文: {encrypted.hex()}")
print(f"解密: {decrypted}")
```

### 1.2 非对称加密

非对称加密解决了密钥分发问题，使用公钥和私钥对进行加密和解密。

**核心思想**：
- **公钥**：公开发布，用于加密或验证签名
- **私钥**：保密存储，用于解密或创建签名
- 数学关系：用公钥加密的数据只能用对应的私钥解密

**应用场景**：
1. **密钥交换**：使用RSA或Diffie-Hellman交换对称密钥
2. **数字签名**：验证消息来源和完整性
3. **身份认证**：SSL/TLS握手、SSH登录

```python
# RSA非对称加密示例
from cryptography.hazmat.primitives import serialization, hashes
from cryptography.hazmat.primitives.asymmetric import rsa, padding

# 生成RSA密钥对
private_key = rsa.generate_private_key(
    public_exponent=65537,
    key_size=2048,
)
public_key = private_key.public_key()

# 公钥加密
def rsa_encrypt(plaintext, public_key):
    ciphertext = public_key.encrypt(
        plaintext.encode('utf-8'),
        padding.OAEP(
            mgf=padding.MGF1(algorithm=hashes.SHA256()),
            algorithm=hashes.SHA256(),
            label=None
        )
    )
    return ciphertext

# 私钥解密
def rsa_decrypt(ciphertext, private_key):
    plaintext = private_key.decrypt(
        ciphertext,
        padding.OAEP(
            mgf=padding.MGF1(algorithm=hashes.SHA256()),
            algorithm=hashes.SHA256(),
            label=None
        )
    )
    return plaintext.decode('utf-8')

# 使用示例
message = "这是一个RSA加密测试"
encrypted_msg = rsa_encrypt(message, public_key)
decrypted_msg = rsa_decrypt(encrypted_msg, private_key)

print(f"原始消息: {message}")
print(f"加密后长度: {len(encrypted_msg)}字节")
print(f"解密结果: {decrypted_msg}")
```

### 1.3 哈希函数

哈希函数将任意长度的输入转换为固定长度的输出，具有单向性和抗碰撞性。

**核心特性**：
- **确定性**：相同输入总是产生相同输出
- **快速计算**：对任意输入都能快速计算哈希值
- **抗原像攻击**：给定哈希值h，难以找到输入m使得H(m)=h
- **抗第二原像攻击**：给定输入m1，难以找到不同的m2使得H(m1)=H(m2)
- **抗碰撞性**：难以找到两个不同的输入m1和m2使得H(m1)=H(m2)

**区块链中的应用**：
- 区块链接：每个区块包含前一个区块的哈希值
- 交易ID：交易数据的唯一标识
- 工作量证明：比特币挖矿就是寻找特定哈希值的过程
- 默克尔树：高效验证大量交易

```python
# SHA-256哈希函数示例
import hashlib
import json

def calculate_hash(data):
    """计算数据的SHA-256哈希值"""
    if isinstance(data, str):
        data = data.encode('utf-8')
    return hashlib.sha256(data).hexdigest()

def create_block_hash(block_data):
    """创建区块哈希示例"""
    # 模拟区块数据结构
    block = {
        "index": block_data["index"],
        "timestamp": block_data["timestamp"],
        "transactions": block_data["transactions"],
        "previous_hash": block_data["previous_hash"],
        "nonce": block_data["nonce"]
    }
    
    # 将区块数据转换为字符串并计算哈希
    block_string = json.dumps(block, sort_keys=True)
    return calculate_hash(block_string)

# 演示比特币区块哈希计算
block_data = {
    "index": 100000,
    "timestamp": "2021-05-22 18:30:00",
    "transactions": [
        {"from": "Alice", "to": "Bob", "amount": 1.5},
        {"from": "Charlie", "to": "David", "amount": 0.8}
    ],
    "previous_hash": "0000000000000000000a3d9e4e9f2e7d5c6b4a2f1e3d5c7b9a0f2e4d6c8b0a",
    "nonce": 0
}

# 模拟挖矿过程 - 寻找前导零
target = "0000"  # 简化的目标
nonce = 0
while True:
    block_data["nonce"] = nonce
    block_hash = create_block_hash(block_data)
    
    if block_hash.startswith(target):
        print(f"找到有效哈希！")
        print(f"区块哈希: {block_hash}")
        print(f"使用的nonce: {nonce}")
        print(f"前导零数量: {target.count('0')}")
        break
    
    nonce += 1
    if nonce > 1000000:  # 防止无限循环
        print("未在限制内找到有效哈希")
        break
```

### 1.4 数字签名

数字签名结合了非对称加密和哈希函数，提供消息认证、完整性和不可否认性。

**签名过程**：
1. 计算消息的哈希值H(M)
2. 使用私钥加密哈希值得到签名S = E_{SK}(H(M))
3. 将消息M和签名S一起发送

**验证过程**：
1. 接收方计算收到消息的哈希值H(M')
2. 使用公钥解密签名得到H(M) = D_{PK}(S)
3. 比较H(M')和H(M)，如果相同则验证成功

```python
# ECDSA数字签名示例（比特币使用这种签名算法）
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.asymmetric import ec
from cryptography.hazmat.backends import default_backend

def generate_ecdsa_keys():
    """生成ECDSA密钥对"""
    private_key = ec.generate_private_key(
        ec.SECP256K1(),  # 比特币使用的曲线
        default_backend()
    )
    public_key = private_key.public_key()
    return private_key, public_key

def sign_message(message, private_key):
    """使用私钥签名消息"""
    if isinstance(message, str):
        message = message.encode('utf-8')
    
    signature = private_key.sign(
        message,
        ec.ECDSA(hashes.SHA256())
    )
    return signature

def verify_signature(message, signature, public_key):
    """使用公钥验证签名"""
    if isinstance(message, str):
        message = message.encode('utf-8')
    
    try:
        public_key.verify(
            signature,
            message,
            ec.ECDSA(hashes.SHA256())
        )
        return True
    except:
        return False

# 使用示例
private_key, public_key = generate_ecdsa_keys()

# 签名交易
transaction = "Alice向Bob转账1.5BTC"
signature = sign_message(transaction, private_key)

# 验证签名
is_valid = verify_signature(transaction, signature, public_key)

print(f"交易内容: {transaction}")
print(f"签名长度: {len(signature)}字节")
print(f"签名验证: {'成功' if is_valid else '失败'}")

# 尝试篡改后验证
tampered_transaction = "Alice向Bob转账15BTC"  # 篡改金额
is_valid_after_tamper = verify_signature(tampered_transaction, signature, public_key)
print(f"篡改后验证: {'成功' if is_valid_after_tamper else '失败'}")
```

## 第二部分：常见加密算法详解

### 2.1 AES（高级加密标准）

AES是目前最广泛使用的对称加密算法，替代了DES算法。

**技术规格**：
- **分组大小**：128位
- **密钥长度**：128、192、256位（AES-128、AES-192、AES-256）
- **轮数**：10轮（128位）、12轮（192位）、14轮（256位）

**加密过程**：
1. **初始轮**：AddRoundKey（轮密钥加）
2. **中间轮**：SubBytes（字节替换）→ ShiftRows（行移位）→ MixColumns（列混合）→ AddRoundKey
3. **最终轮**：SubBytes → ShiftRows → AddRoundKey（无列混合）

**安全性分析**：
- **理论安全性**：128位密钥提供2^128种可能性，暴力破解不可行
- **实际攻击**：侧信道攻击（时间、功耗、缓存攻击）
- **量子威胁**：Grover算法将有效密钥长度减半，AES-256仍安全

### 2.2 RSA算法

RSA是第一个也是最流行的公钥加密算法，基于大数分解难题。

**数学原理**：
1. 选择两个大质数p和q
2. 计算n = p × q（模数）
3. 计算φ(n) = (p-1)(q-1)（欧拉函数）
4. 选择公钥指数e，满足1 < e < φ(n)且gcd(e, φ(n)) = 1
5. 计算私钥指数d，满足d × e ≡ 1 mod φ(n)

**性能特点**：
- **加密速度慢**：相比AES慢1000倍以上
- **适合小数据量**：通常用于加密对称密钥或签名
- **密钥长度长**：2048位RSA提供约112位安全性

### 2.3 ECC（椭圆曲线密码学）

ECC基于椭圆曲线离散对数问题，在相同安全强度下需要更短的密钥。

**优势**：
- **密钥长度短**：256位ECC ≈ 3072位RSA的安全性
- **计算效率高**：更快的加密/解密、签名/验证
- **带宽节省**：适合移动设备和IoT

**比特币中的ECC**：
- 使用secp256k1曲线：y² = x³ + 7 mod p
- 私钥：256位随机数
- 公钥：曲线上的点（x, y坐标）
- 地址生成：公钥 → SHA-256 → RIPEMD-160 → Base58Check编码

### 2.4 SHA-256算法

SHA-256是SHA-2家族中的256位哈希算法，比特币的工作量证明算法。

**算法流程**：
1. **消息填充**：使消息长度 ≡ 448 mod 512
2. **解析消息**：分成512位的块
3. **初始化哈希值**：8个32位寄存器（H0-H7）
4. **处理每个块**：进行64轮运算
5. **最终哈希**：连接H0-H7得到256位输出

**安全性**：
- 抗原像攻击：2^256次操作
- 抗第二原像攻击：2^128次操作
- 抗碰撞性：2^128次操作（生日攻击）

## 第三部分：密钥管理

### 3.1 KMS（密钥管理服务）

云服务提供商提供的托管密钥管理解决方案。

**主要功能**：
- **密钥生成**：使用硬件安全模块生成强随机密钥
- **密钥存储**：安全存储密钥，支持自动轮换
- **加密/解密服务**：提供API接口处理加密操作
- **审计日志**：记录所有密钥使用活动

**AWS KMS示例**：
```python
import boto3
from botocore.exceptions import ClientError

class AWSKMSManager:
    def __init__(self, region_name='us-east-1'):
        self.kms_client = boto3.client('kms', region_name=region_name)
    
    def create_key(self, key_description):
        """创建KMS主密钥"""
        try:
            response = self.kms_client.create_key(
                Description=key_description,
                KeyUsage='ENCRYPT_DECRYPT',
                Origin='AWS_KMS'
            )
            return response['KeyMetadata']['KeyId']
        except ClientError as e:
            print(f"创建密钥失败: {e}")
            return None
    
    def encrypt_data(self, key_id, plaintext):
        """使用KMS加密数据"""
        try:
            response = self.kms_client.encrypt(
                KeyId=key_id,
                Plaintext=plaintext.encode('utf-8')
            )
            return response['CiphertextBlob']
        except ClientError as e:
            print(f"加密失败: {e}")
            return None
    
    def decrypt_data(self, ciphertext_blob):
        """使用KMS解密数据"""
        try:
            response = self.kms_client.decrypt(
                CiphertextBlob=ciphertext_blob
            )
            return response['Plaintext'].decode('utf-8')
        except ClientError as e:
            print(f"解密失败: {e}")
            return None
    
    def enable_key_rotation(self, key_id):
        """启用自动密钥轮换"""
        try:
            self.kms_client.enable_key_rotation(KeyId=key_id)
            return True
        except ClientError as e:
            print(f"启用密钥轮换失败: {e}")
            return False

# 使用示例
kms = AWSKMSManager()
key_id = kms.create_key("区块链交易加密密钥")
if key_id:
    encrypted = kms.encrypt_data(key_id, "私钥数据")
    decrypted = kms.decrypt_data(encrypted)
    print(f"加密后: {encrypted}")
    print(f"解密后: {decrypted}")
```

### 3.3 密钥轮换策略

定期更换密钥以减少密钥泄露的风险。

**最佳实践**：
1. **定期轮换**：根据安全策略（如90天）自动轮换
2. **事件触发轮换**：检测到可疑活动或合规要求时轮换
3. **平滑过渡**：新旧密钥共存一段时间，确保业务连续性
4. **安全销毁**：彻底删除旧密钥，防止恢复

```python
# 密钥轮换管理器示例
import datetime
import hashlib
import json

class KeyRotationManager:
    def __init__(self):
        self.keys = {}  # key_id: {"key": key, "created": datetime, "expires": datetime}
    
    def generate_key(self):
        """生成新密钥"""
        import os
        return os.urandom(32)
    
    def add_key(self, key_id, ttl_days=90):
        """添加新密钥并设置过期时间"""
        key = self.generate_key()
        now = datetime.datetime.now()
        
        self.keys[key_id] = {
            "key": key.hex(),
            "created": now,
            "expires": now + datetime.timedelta(days=ttl_days),
            "active": True
        }
        
        return key
    
    def rotate_key(self, key_id, ttl_days=90):
        """轮换密钥"""
        # 1. 生成新密钥
        new_key = self.add_key(f"{key_id}_new", ttl_days)
        
        # 2. 标记旧密钥为待停用
        if key_id in self.keys:
            self.keys[key_id]["active"] = False
            self.keys[key_id]["deprecation_date"] = datetime.datetime.now()
        
        # 3. 重命名新密钥
        if f"{key_id}_new" in self.keys:
            self.keys[key_id] = self.keys.pop(f"{key_id}_new")
        
        return new_key
    
    def check_and_rotate(self):
        """检查并轮换过期的密钥"""
        now = datetime.datetime.now()
        keys_to_rotate = []
        
        for key_id, key_info in self.keys.items():
            if key_info["expires"] < now:
                keys_to_rotate.append(key_id)
        
        for key_id in keys_to_rotate:
            print(f"轮换过期密钥: {key_id}")
            self.rotate_key(key_id)
        
        return len(keys_to_rotate)
    
    def get_active_key(self, key_id):
        """获取活跃密钥"""
        if key_id in self.keys and self.keys[key_id]["active"]:
            return self.keys[key_id]["key"]
        return None

# 使用示例
manager = KeyRotationManager()
manager.add_key("transaction_signing_key")
print(f"原始密钥: {manager.get_active_key('transaction_signing_key')[:20]}...")

# 模拟密钥轮换
rotated_key = manager.rotate_key("transaction_signing_key")
print(f"轮换后密钥: {rotated_key.hex()[:20]}...")
```

## 第四部分：TLS/SSL协议

### 4.1 TLS握手过程

TLS（传输层安全）协议为网络通信提供安全通道。

**TLS 1.3握手流程**：
1. **ClientHello**：客户端发送支持的密码套件、随机数、扩展
2. **ServerHello**：服务器选择密码套件、发送随机数、证书
3. **密钥交换**：服务器发送密钥交换参数（如ECDHE）
4. **证书验证**：客户端验证服务器证书
5. **Finished**：双方发送Finished消息，握手完成

**主要改进**（相比TLS 1.2）：
- **更快握手**：1-RTT甚至0-RTT恢复
- **移除不安全特性**：静态RSA、CBC模式密码套件
- **前向保密**：必须支持ECDHE或DHE

```python
# Python SSL/TLS客户端示例
import ssl
import socket
import urllib.request

def check_tls_connection(hostname, port=443):
    """检查TLS连接和证书信息"""
    context = ssl.create_default_context()
    
    try:
        with socket.create_connection((hostname, port)) as sock:
            with context.wrap_socket(sock, server_hostname=hostname) as ssock:
                # 获取证书信息
                cert = ssock.getpeercert()
                cipher = ssock.cipher()
                version = ssock.version()
                
                print(f"TLS版本: {version}")
                print(f"密码套件: {cipher[0]}")
                print(f"证书主题: {cert['subject']}")
                print(f"证书颁发者: {cert['issuer']}")
                print(f"有效期: {cert['notBefore']} 到 {cert['notAfter']}")
                
                # 检查证书扩展
                if 'subjectAltName' in cert:
                    print(f"备用域名: {cert['subjectAltName']}")
                
                return True
                
    except ssl.SSLError as e:
        print(f"SSL错误: {e}")
        return False
    except socket.error as e:
        print(f"连接错误: {e}")
        return False
```

### 5.2 共识机制

共识机制确保分布式网络中节点对账本状态达成一致。

**工作量证明（PoW）**：
- **原理**：节点通过计算哈希难题竞争记账权
- **特点**：安全性高、能耗大、确认慢
- **应用**：比特币、莱特币、以太坊（历史）

**权益证明（PoS）**：
- **原理**：节点根据持有的代币数量和质押时间获得记账权
- **特点**：能耗低、可扩展性好、富者更富
- **应用**：以太坊2.0、Cardano、Polkadot

**委托权益证明（DPoS）**：
- **原理**：代币持有者投票选举代表进行记账
- **特点**：高吞吐量、中心化风险
- **应用**：EOS、Tron

### 5.3 P2P网络

区块链使用对等网络进行节点间通信。

**网络拓扑**：
- **非结构化网络**：节点随机连接，简单但效率低
- **结构化网络**：基于DHT（分布式哈希表）组织，如Kademlia
- **比特币网络**：基于非结构化P2P，使用gossip协议传播交易

**节点类型**：
- **全节点**：存储完整区块链，验证所有交易
- **轻节点**：只存储区块头，依赖全节点获取详细信息
- **挖矿节点**：参与共识过程，创建新区块

```python
# 简化的区块链实现
import hashlib
import json
import time
from typing import List, Dict

class Block:
    def __init__(self, index: int, transactions: List[Dict], 
                 previous_hash: str, nonce: int = 0):
        self.index = index
        self.timestamp = time.time()
        self.transactions = transactions
        self.previous_hash = previous_hash
        self.nonce = nonce
        self.hash = self.calculate_hash()
    
    def calculate_hash(self) -> str:
        block_data = {
            "index": self.index,
            "timestamp": self.timestamp,
            "transactions": self.transactions,
            "previous_hash": self.previous_hash,
            "nonce": self.nonce
        }
        block_string = json.dumps(block_data, sort_keys=True)
        return hashlib.sha256(block_string.encode()).hexdigest()
    
    def mine_block(self, difficulty: int):
        target = "0" * difficulty
        while self.hash[:difficulty] != target:
            self.nonce += 1
            self.hash = self.calculate_hash()
        print(f"区块 {self.index} 已挖出: {self.hash}")

class Blockchain:
    def __init__(self):
        self.chain: List[Block] = []
        self.difficulty = 4
        self.pending_transactions: List[Dict] = []
        self.mining_reward = 10
        
        # 创建创世区块
        self.create_genesis_block()
    
    def create_genesis_block(self):
        genesis_block = Block(0, [], "0")
        genesis_block.mine_block(self.difficulty)
        self.chain.append(genesis_block)
    
    def get_latest_block(self) -> Block:
        return self.chain[-1]
    
    def add_transaction(self, transaction: Dict):
        self.pending_transactions.append(transaction)
    
    def mine_pending_transactions(self, miner_address: str):
        # 添加挖矿奖励交易
        reward_tx = {
            "sender": "network",
            "recipient": miner_address,
            "amount": self.mining_reward
        }
        self.pending_transactions.append(reward_tx)
        
        # 创建新区块
        new_block = Block(
            index=len(self.chain),
            transactions=self.pending_transactions,
            previous_hash=self.get_latest_block().hash
        )
        
        # 挖矿
        new_block.mine_block(self.difficulty)
        
        # 添加到链
        self.chain.append(new_block)
        self.pending_transactions = []
        
        return new_block
    
    def is_chain_valid(self) -> bool:
        for i in range(1, len(self.chain)):
            current_block = self.chain[i]
            previous_block = self.chain[i-1]
            
            # 验证当前区块哈希
            if current_block.hash != current_block.calculate_hash():
                return False
            
            # 验证链接
            if current_block.previous_hash != previous_block.hash:
                return False
        
        return True

# 使用示例
blockchain = Blockchain()

# 添加交易
blockchain.add_transaction({"sender": "Alice", "recipient": "Bob", "amount": 5})
blockchain.add_transaction({"sender": "Bob", "recipient": "Charlie", "amount": 2.5})

# 挖矿
print("开始挖矿...")
mined_block = blockchain.mine_pending_transactions("Miner1")
print(f"区块已添加，哈希: {mined_block.hash}")

# 验证区块链
print(f"区块链有效性: {blockchain.is_chain_valid()}")
```

## 第六部分：比特币技术详解

### 6.1 UTXO模型

比特币使用未花费交易输出（UTXO）模型跟踪余额。

**核心概念**：
- **UTXO**：不可分割的交易输出，类似于现金中的纸币
- **交易输入**：引用之前交易的UTXO
- **交易输出**：创建新的UTXO
- **余额计算**：所有可花费UTXO的总和

**与账户模型对比**：
- **UTXO模型**：比特币、莱特币、比特币现金
  - 优点：隐私性好、并行处理、简单验证
  - 缺点：状态管理复杂、需要找零机制
- **账户模型**：以太坊、大多数智能合约平台
  - 优点：编程简单、状态明确
  - 缺点：隐私性差、状态膨胀

```python
# 简化的UTXO模型实现
from typing import Dict, List, Set, Tuple
import hashlib

class UTXO:
    def __init__(self, tx_id: str, output_index: int, amount: float, 
                 script_pubkey: str):
        self.tx_id = tx_id
        self.output_index = output_index
        self.amount = amount
        self.script_pubkey = script_pubkey  # 锁定脚本
    
    def to_dict(self) -> Dict:
        return {
            "tx_id": self.tx_id,
            "output_index": self.output_index,
            "amount": self.amount,
            "script_pubkey": self.script_pubkey
        }

class Transaction:
    def __init__(self, inputs: List[Dict], outputs: List[Dict]):
        self.inputs = inputs  # [{utxo: UTXO, script_sig: str}]
        self.outputs = outputs  # [{amount: float, script_pubkey: str}]
        self.tx_id = self.calculate_txid()
    
    def calculate_txid(self) -> str:
        tx_data = json.dumps({
            "inputs": self.inputs,
            "outputs": self.outputs
        }, sort_keys=True)
        return hashlib.sha256(tx_data.encode()).hexdigest()

class UTXOSet:
    def __init__(self):
        self.utxos: Dict[str, UTXO] = {}  # key: "txid:output_index"
    
    def add_utxo(self, utxo: UTXO):
        key = f"{utxo.tx_id}:{utxo.output_index}"
        self.utxos[key] = utxo
    
    def remove_utxo(self, tx_id: str, output_index: int):
        key = f"{tx_id}:{output_index}"
        if key in self.utxos:
            del self.utxos[key]
    
    def get_utxo(self, tx_id: str, output_index: int) -> UTXO:
        key = f"{tx_id}:{output_index}"
        return self.utxos.get(key)
    
    def get_balance(self, address: str) -> float:
        """计算地址余额"""
        balance = 0.0
        for utxo in self.utxos.values():
            if utxo.script_pubkey == f"OP_DUP OP_HASH160 {address} OP_EQUALVERIFY OP_CHECKSIG":
                balance += utxo.amount
        return balance
    
    def get_utxos_for_address(self, address: str) -> List[UTXO]:
        """获取地址的所有UTXO"""
        utxos = []
        for utxo in self.utxos.values():
            if utxo.script_pubkey == f"OP_DUP OP_HASH160 {address} OP_EQUALVERIFY OP_CHECKSIG":
                utxos.append(utxo)
        return utxos

# 使用示例
utxo_set = UTXOSet()

# 创建初始UTXO
genesis_utxo = UTXO(
    tx_id="genesis_tx",
    output_index=0,
    amount=50.0,
    script_pubkey="OP_DUP OP_HASH160 1A1zP1eP5QGefi2DMPTfTL5SLmv7DivfNa OP_EQUALVERIFY OP_CHECKSIG"
)
utxo_set.add_utxo(genesis_utxo)

# 查询余额
balance = utxo_set.get_balance("1A1zP1eP5QGefi2DMPTfTL5SLmv7DivfNa")
print(f"地址余额: {balance} BTC")
```
