# 系统设计面试完全指南

## 一、系统设计基本方法论

### 1.1 完整设计流程框架

系统设计面试考察的是系统性思考能力，通常遵循以下四步法：

```
需求分析 → 高层设计 → 详细设计 → 扩展优化
```

#### 需求分析阶段（5-10分钟）
```python
# 示例：设计一个短链服务的需求分析清单
class RequirementsAnalyzer:
    def __init__(self):
        self.functional_reqs = []
        self.non_functional_reqs = {}
        self.constraints = []
        self.assumptions = []
    
    def analyze(self, problem_statement):
        # 1. 明确核心功能
        self.functional_reqs = [
            "将长URL转换为短URL",
            "用户访问短URL时重定向到原始URL", 
            "支持自定义短链（可选）",
            "提供访问统计（可选）"
        ]
        
        # 2. 确定非功能性需求
        self.non_functional_reqs = {
            "可用性": "99.99%",
            "延迟": "< 100ms 重定向时间",
            "吞吐量": "1000 QPS 写入，10000 QPS 读取",
            "数据持久化": "永久存储",
            "一致性": "最终一致性可接受"
        }
        
        # 3. 明确约束条件
        self.constraints = [
            "短码长度6-8位",
            "支持字母数字混合",
            "防滥用机制"
        ]
        
        # 4. 假设和澄清
        self.assumptions = [
            "读写比例大约 10:1",
            "URL平均长度1000字符",
            "系统需要支撑5年数据"
        ]
```

深入每个组件的实现细节，包括数据模型、算法选择、接口设计等。

考虑扩展性、容错性、监控等运维层面设计。

### 1.2 面试中的沟通策略

```markdown
1. **结构化表达**：使用"首先...其次...最后..."的结构
2. **可视化思维**：边说边画架构图
3. **权衡分析**：每个选择都说明优缺点
4. **互动确认**：定期确认面试官期望

```

class OrderService:
    def __init__(self, order_repo, inventory_service, payment_service):
        self.order_repo = order_repo
        self.inventory_service = inventory_service
        self.payment_service = payment_service
    
    def create_order(self, order_data):
        # 1. 检查库存
        if not self.inventory_service.check_stock(order_data['items']):
            raise InsufficientStockError()
        
        # 2. 创建订单
        order = Order(
            user_id=order_data['user_id'],
            items=order_data['items'],
            total_amount=self.calculate_total(order_data['items'])
        )
        
        # 3. 预扣库存
        self.inventory_service.reserve_stock(order.items)
        
        # 4. 发起支付
        payment_result = self.payment_service.initiate_payment(
            order.id, order.total_amount
        )
        
        # 5. 保存订单
        saved_order = self.order_repo.save(order)
        
        return {
            'order_id': saved_order.id,
            'payment_url': payment_result['payment_url']
        }

# 数据访问层
class OrderRepository:
    def __init__(self, db_session):
        self.db = db_session
    
    def save(self, order):
        order_entity = OrderEntity(
            id=order.id,
            user_id=order.user_id,
            status=order.status.value,
            created_at=datetime.utcnow()
        )
        
        # 保存订单项
        for item in order.items:
            item_entity = OrderItemEntity(
                order_id=order.id,
                product_id=item.product_id,
                quantity=item.quantity,
                price=item.price
            )
            self.db.add(item_entity)
        
        self.db.add(order_entity)
        self.db.commit()
        
        return self.to_domain(order_entity)
```

#### 分层架构的优缺点
```
优点：
1. 关注点分离，降低耦合度
2. 易于理解和维护
3. 支持并行开发
4. 层次间可独立替换

缺点：
1. 层间调用性能开销
2. 过度设计简单业务
3. 可能导致"土堆"反模式
```

# 订单事件溯源实现
class OrderEventStore:
    def __init__(self, event_store_db):
        self.db = event_store_db
    
    def save_event(self, aggregate_id, event):
        event_record = {
            'aggregate_id': aggregate_id,
            'event_type': type(event).__name__,
            'event_data': event.to_dict(),
            'version': self.get_next_version(aggregate_id),
            'timestamp': datetime.utcnow()
        }
        self.db.events.insert_one(event_record)
    
    def get_events(self, aggregate_id):
        return list(self.db.events.find(
            {'aggregate_id': aggregate_id}
        ).sort('version', 1))

class Order:
    def __init__(self, order_id):
        self.id = order_id
        self.version = 0
        self.status = 'CREATED'
        self.items = []
        self.uncommitted_events = []
    
    def add_item(self, product_id, quantity, price):
        # 业务规则验证
        if self.status != 'CREATED':
            raise Exception("Cannot modify order in current status")
        
        # 创建事件
        event = OrderItemAdded(
            order_id=self.id,
            product_id=product_id,
            quantity=quantity,
            price=price
        )
        
        # 应用事件
        self.apply(event)
        self.uncommitted_events.append(event)
    
    def apply(self, event):
        if isinstance(event, OrderItemAdded):
            self.items.append({
                'product_id': event.product_id,
                'quantity': event.quantity,
                'price': event.price
            })
        
        self.version += 1
```

# 基于Redis的发布订阅实现
import redis
import json
import threading

class EventPublisher:
    def __init__(self, redis_client):
        self.redis = redis_client
    
    def publish(self, channel, event):
        event_data = {
            'event_id': str(uuid.uuid4()),
            'timestamp': datetime.utcnow().isoformat(),
            'data': event
        }
        self.redis.publish(channel, json.dumps(event_data))

class EventSubscriber:
    def __init__(self, redis_client):
        self.redis = redis_client
        self.handlers = {}
    
    def subscribe(self, channel, handler):
        if channel not in self.handlers:
            self.handlers[channel] = []
            # 启动监听线程
            threading.Thread(
                target=self._listen_channel,
                args=(channel,),
                daemon=True
            ).start()
        
        self.handlers[channel].append(handler)
    
    def _listen_channel(self, channel):
        pubsub = self.redis.pubsub()
        pubsub.subscribe(channel)
        
        for message in pubsub.listen():
            if message['type'] == 'message':
                event_data = json.loads(message['data'])
                for handler in self.handlers[channel]:
                    try:
                        handler(event_data)
                    except Exception as e:
                        print(f"Handler error: {e}")

# 使用示例
redis_client = redis.Redis()
publisher = EventPublisher(redis_client)
subscriber = EventSubscriber(redis_client)

# 订单服务发布事件
publisher.publish('order-events', {
    'order_id': '12345',
    'event_type': 'ORDER_CREATED',
    'amount': 100.00
})

# 库存服务订阅事件
def handle_order_created(event):
    if event['data']['event_type'] == 'ORDER_CREATED':
        # 扣减库存
        inventory_service.reserve_stock(event['data']['order_id'])

subscriber.subscribe('order-events', handle_order_created)

# CQRS完整实现示例 - 用户管理系统
# 命令端 - 处理写操作
class UserCommandHandler:
    def __init__(self, event_store, event_publisher):
        self.event_store = event_store
        self.event_publisher = event_publisher
    
    def handle_create_user(self, command):
        # 验证业务规则
        if self.user_exists(command.email):
            raise UserAlreadyExistsError()
        
        # 创建聚合根
        user = User.create(
            user_id=str(uuid.uuid4()),
            email=command.email,
            name=command.name
        )
        
        # 保存事件
        for event in user.uncommitted_events:
            self.event_store.save_event(user.id, event)
        
        # 发布事件用于查询端更新
        self.event_publisher.publish('user-events', {
            'events': [e.to_dict() for e in user.uncommitted_events]
        })
        
        return user.id

# 查询端 - 处理读操作
class UserQueryHandler:
    def __init__(self, read_model_db):
        self.db = read_model_db
    
    def get_user_profile(self, user_id):
        # 从读模型获取，可能包含冗余数据
        return self.db.user_profiles.find_one({'user_id': user_id})
    
    def search_users(self, query, page, page_size):
        # 全文搜索，使用专门优化的数据结构
        return self.db.user_search.aggregate([
            {'$match': {'$text': {'$search': query}}},
            {'$skip': page * page_size},
            {'$limit': page_size},
            {'$project': {'user_id': 1, 'name': 1, 'email': 1, 'score': 1}}
        ])

# 事件处理器更新查询端
class UserProfileProjection:
    def __init__(self, read_model_db, event_subscriber):
        self.db = read_model_db
        self.event_subscriber = event_subscriber
        self.event_subscriber.subscribe('user-events', self.handle_user_events)
    
    def handle_user_events(self, event_data):
        for event in event_data['events']:
            if event['type'] == 'USER_CREATED':
                self.handle_user_created(event)
            elif event['type'] == 'USER_UPDATED':
                self.handle_user_updated(event)
    
    def handle_user_created(self, event):

            'user_id': event['user_id'],
            'name': event['name'],
            'email': event['email'],
            'created_at': event['timestamp'],
            'profile_views': 0,
            'search_score': 0

        # 更新搜索索引

#### CQRS的优势与挑战
```
优势：
1. 读写模型独立优化
2. 针对不同场景使用不同数据存储
3. 提高查询性能
4. 更好的扩展性

挑战：
1. 系统复杂度增加
2. 数据一致性维护复杂
3. 开发和维护成本高
4. 需要事件驱动架构支持
```

class DatabaseSelector:
    def __init__(self):
        self.decision_tree = {
            'data_structure': {
                'structured': {
                    'consistency': 'strong',
                    'transaction': 'required',
                    'recommendation': 'SQL (MySQL/PostgreSQL)'
                },
                'semi_structured': {
                    'schema_flexibility': 'high',
                    'scalability': 'horizontal',
                    'recommendation': 'Document DB (MongoDB)'
                },
                'unstructured': {
                    'query_pattern': 'simple',
                    'write_intensive': True,
                    'recommendation': 'Key-Value (Redis/DynamoDB)'
                }
            }
        }
    
    def select_database(self, requirements):
        # 分析需求
        data_type = self.analyze_data_type(requirements)
        consistency = self.check_consistency_requirements(requirements)
        scale = self.estimate_scale(requirements)
        
        # 决策逻辑
        if data_type == 'graph' and requirements.get('relationship_complex'):
            return 'Neo4j', 'Graph Database'
        
        if requirements.get('search_complex'):
            return 'Elasticsearch', 'Search Engine'
        
        if requirements.get('time_series'):
            return 'InfluxDB', 'Time Series DB'
        
        if consistency == 'strong' and data_type == 'structured':
            return 'PostgreSQL', 'ACID Compliance'
        
        if scale == 'write_intensive' and data_type == 'simple_kv':
            return 'Redis', 'High Performance Caching'
        
        if scale == 'horizontal' and data_type == 'document':
            return 'MongoDB', 'Flexible Schema'
        
        return 'MySQL', 'Default Choice'
```

# 基于用户ID的分片策略
class ShardManager:
    def __init__(self, shard_config):
        self.shard_config = shard_config
        self.shards = self._initialize_shards()
    
    def _initialize_shards(self):
        shards = {}
        for shard_id, config in self.shard_config.items():
            shards[shard_id] = {
                'connection': self._create_connection(config),
                'range': config['range']
            }
        return shards
    
    def get_shard_for_user(self, user_id):
        """基于用户ID的哈希分片"""
        # 方法1: 取模分片
        shard_key = user_id % self.shard_config['total_shards']
        
        # 方法2: 一致性哈希
        # shard_key = self._consistent_hash(user_id)
        
        return self.shards[shard_key]['connection']
    
    def get_shard_for_range(self, start_time, end_time):
        """基于时间范围的分片"""
        # 按月分片示例
        start_month = start_time.strftime('%Y%m')
        end_month = end_time.strftime('%Y%m')
        
        shards_needed = []
        current = start_month
        while current <= end_month:
            if current in self.shards:
                shards_needed.append(self.shards[current]['connection'])
            # 下个月
            current = self._next_month(current)
        
        return shards_needed

# 分片键选择评估框架
class ShardKeyEvaluator:
    def evaluate_candidate_keys(self, candidate_keys, data_distribution):
        results = {}
        
        for key in candidate_keys:
            score = 0
            
            # 1. 数据分布均匀性 (权重: 40%)
            uniformity_score = self._calculate_uniformity(
                data_distribution[key]
            )
            score += uniformity_score * 0.4
            
            # 2. 查询局部性 (权重: 30%)
            locality_score = self._calculate_query_locality(key)
            score += locality_score * 0.3
            
            # 3. 跨分片查询频率 (权重: 20%)
            cross_shard_score = self._calculate_cross_shard_queries(key)
            score += cross_shard_score * 0.2
            
            # 4. 更新频率 (权重: 10%)
            update_frequency = self._calculate_update_frequency(key)
            score += update_frequency * 0.1
            
            results[key] = {
                'total_score': score,
                'details': {
                    'uniformity': uniformity_score,
                    'locality': locality_score,
                    'cross_shard': cross_shard_score,
                    'update_freq': update_frequency
                }
            }
        
        return sorted(results.items(), key=lambda x: x[1]['total_score'])
    
    def _calculate_uniformity(self, distribution):
        """计算数据分布的均匀性"""
        # 使用基尼系数或标准差
        values = list(distribution.values())
        if not values:
            return 0
        
        mean = sum(values) / len(values)
        variance = sum((x - mean) ** 2 for x in values) / len(values)
        
        # 返回均匀性分数 (1 - 标准化标准差)
        return 1 - (variance ** 0.5 / mean) if mean > 0 else 0
```

# 读写分离中间件实现
import random
from contextlib import contextmanager

class ReadWriteSplitter:
    def __init__(self, master_config, slave_configs):
        self.master_pool = self._create_connection_pool(master_config)
        self.slave_pools = [self._create_connection_pool(config) 
                           for config in slave_configs]
        self.slave_index = 0
    
    @contextmanager
    def master_connection(self):
        """获取主库连接用于写操作"""
        conn = self.master_pool.get_connection()
        try:
            yield conn
            conn.commit()
        except Exception as e:
            conn.rollback()
            raise e
        finally:
            conn.close()
    
    @contextmanager
    def slave_connection(self):
        """获取从库连接用于读操作"""
        # 简单轮询策略
        pool = self.slave_pools[self.slave_index % len(self.slave_pools)]
        self.slave_index += 1
        
        conn = pool.get_connection()
        try:
            yield conn
        finally:
            conn.close()
    
    def execute_write(self, query, params=None):
        """执行写操作"""
        with self.master_connection() as conn:
            cursor = conn.cursor()
            cursor.execute(query, params)
            return cursor.lastrowid
    
    def execute_read(self, query, params=None):
        """执行读操作"""
        with self.slave_connection() as conn:
            cursor = conn.cursor()
            cursor.execute(query, params)
            return cursor.fetchall()
    
    def execute_with_fallback(self, query, params=None, allow_stale=False):
        """带降级策略的读操作"""
        try:
            # 先尝试从库
            if not allow_stale:
                return self.execute_read(query, params)
            else:
                # 如果允许读旧数据，可以设置较短的超时
                return self._execute_with_timeout(
                    self.execute_read, 
                    query, 
                    params, 
                    timeout=0.1
                )
        except Exception as e:
            # 从库失败，降级到主库
            print(f"Slave read failed: {e}, falling back to master")
            return self.execute_read_on_master(query, params)

import time
import json
from functools import wraps

class MultiLevelCache:
    def __init__(self):
        self.l1_cache = {}  # 本地内存缓存
        self.l2_cache = RedisCache()  # Redis分布式缓存
        self.l3_cache = CDNCache()  # CDN缓存
        self.cache_stats = {
            'l1_hits': 0,
            'l2_hits': 0,
            'l3_hits': 0,
            'misses': 0
        }
    
    def get(self, key):
        # L1: 本地内存 (最快，TTL短)
        if key in self.l1_cache:
            entry = self.l1_cache[key]
            if entry['expire'] > time.time():
                self.cache_stats['l1_hits'] += 1
                return entry['value']
            else:
                del self.l1_cache[key]
        
        # L2: Redis (分布式，中等速度)
        value = self.l2_cache.get(key)
        if value is not None:
            self.cache_stats['l2_hits'] += 1
            # 回填L1
            self.l1_cache[key] = {
                'value': value,
                'expire': time.time() + 60  # 60秒本地缓存
            }
            return value
        
        # L3: CDN (边缘缓存，适合静态内容)
        # 这里简化处理，实际CDN缓存由HTTP层处理
        
        # 全部未命中
        self.cache_stats['misses'] += 1
        return None
    
    def set(self, key, value, ttl=300):
        # 写入L1
        self.l1_cache[key] = {
            'value': value,
            'expire': time.time() + min(ttl, 60)  # 本地缓存不超过60秒
        }
        
        # 写入L2
        self.l2_cache.set(key, value, ttl)
        
        # L3由HTTP缓存头控制
        return True
    
    def delete(self, key):
        # 删除所有层次的缓存
        if key in self.l1_cache:
            del self.l1_cache[key]
        self.l2_cache.delete(key)

# 综合解决方案实现
class RobustCache:
    def __init__(self, redis_client):
        self.redis = redis_client
        self.lock = threading.Lock()
        self.local_cache = {}
        self.bloom_filter = BloomFilter(capacity=1000000)
    
    def get_with_protection(self, key, loader_func, ttl=300):
        """带防穿透保护的缓存读取"""
        # 1. 布隆过滤器检查
        if not self.bloom_filter.check(key):
            return None
        
        # 2. 先查缓存
        value = self.redis.get(key)
        if value:
            # 反序列化
            return json.loads(value)
        
        # 3. 防止缓存击穿 - 使用分布式锁
        lock_key = f"lock:{key}"
        locked = self.redis.set(lock_key, "1", nx=True, ex=5)
        
        if locked:
            try:
                # 双重检查
                value = self.redis.get(key)
                if value:
                    return json.loads(value)
                
                # 加载数据
                value = loader_func()
                
                # 缓存结果（包括空值防穿透）
                if value is None:
                    # 空值缓存较短时间
                    self.redis.setex(key, 60, json.dumps(None))
                else:
                    # 正常缓存
                    self.redis.setex(key, ttl, json.dumps(value))
                    # 更新布隆过滤器
                    self.bloom_filter.add(key)
                
                return value
            finally:
                # 释放锁
                self.redis.delete(lock_key)
        else:
            # 等待其他线程加载
            time.sleep(0.1)
            return self.get_with_protection(key, loader_func, ttl)
    
    def get_with_local_cache(self, key, loader_func, ttl=300, local_ttl=60):
        """带本地缓存的读取"""
        # 1. 本地缓存检查
        if key in self.local_cache:
            entry = self.local_cache[key]
            if entry['expire'] > time.time():
                return entry['value']
        
        # 2. 分布式缓存
        value = self.redis.get(key)
        if value:
            parsed_value = json.loads(value)
            # 更新本地缓存
            self.local_cache[key] = {
                'value': parsed_value,
                'expire': time.time() + local_ttl
            }
            return parsed_value
        
        # 3. 加载并缓存
        value = loader_func()
        
        # 缓存到Redis
        self.redis.setex(key, ttl, json.dumps(value))
        
        # 缓存到本地
        self.local_cache[key] = {
            'value': value,
            'expire': time.time() + local_ttl
        }
        
        return value
    
    def batch_load(self, keys, loader_func, ttl=300):
        """批量加载优化"""
        # 1. 批量查询缓存
        cached_values = {}
        missing_keys = []
        
        # 管道批量查询
        pipe = self.redis.pipeline()
        for key in keys:
            pipe.get(key)
        cache_results = pipe.execute()
        
        for key, value in zip(keys, cache_results):
            if value:
                cached_values[key] = json.loads(value)
            else:
                missing_keys.append(key)
        
        # 2. 批量加载缺失数据
        if missing_keys:
            # 调用批量加载函数
            loaded_values = loader_func(missing_keys)
            
            # 批量写入缓存
            pipe = self.redis.pipeline()
            for key, value in loaded_values.items():
                pipe.setex(key, ttl, json.dumps(value))
                cached_values[key] = value
            pipe.execute()
        
        # 按原始顺序返回结果
        return [cached_values.get(key) for key in keys]
